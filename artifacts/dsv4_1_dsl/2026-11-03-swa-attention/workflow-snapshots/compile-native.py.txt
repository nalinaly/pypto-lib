"""Compile exported MIX roles without allocating an NPU."""

from pathlib import Path
import runpy
import sys

from pypto.runtime.kernel_compiler import KernelCompiler
from simpler_setup import ensure_pto_isa_root


config = runpy.run_path(sys.argv[1])
output = Path(sys.argv[2])
output.mkdir(parents=True, exist_ok=True)
for kernel in config["KERNELS"]:
    compiler = KernelCompiler("a5")
    binary = compiler.compile_incore(
        kernel["source"], core_type=kernel["core_type"],
        pto_isa_root=ensure_pto_isa_root(), runtime_name="tensormap_and_ringbuffer",
        extra_include_dirs=kernel["extra_include_dirs"],
    )
    (output / (kernel["core_type"] + ".elf")).write_bytes(binary)
    print(kernel["core_type"], "COMPILED", len(binary), flush=True)
