"""用原 PyPTO 任务和 CPU 参考逐字节验证 DSL 收集任务的四个实际设备输出。"""

import argparse
import json
from pathlib import Path

import torch
from golden import run

from index_gather_dsl import ROW_TILE, SCALE_COUNT, build_gather_specs, compare_bytes, golden_gather, validate_gather
from native_gather import validate_native


work = Path(__file__).resolve().parent


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("-d", "--device", type=int, default=0)
    parser.add_argument("--compile-only", action="store_true")
    args = parser.parse_args()
    records = []
    for case in ((7, 33, 6), (7, 33, 0), (3, 1, 3), (5, 32, 5), (3, 65, 3), (37, 33, 35), (1, 32, 1, "bytes")):
        record = {"case": case, "输出来源": "compare_fn 的 actual 参数"}
        outputs = []
        for label, entry in (("DSL", validate_gather), ("PyPTO", validate_native)):
            captured = {}

            def capture(name):
                def compare(actual, expected, **kwargs):
                    captured[name] = actual.detach().cpu().clone()
                    return compare_bytes(actual, expected, **kwargs)
                return compare

            print(f"[INDEX_GATHER_COMPARE] {label} {case}", flush=True)
            result = run(
                fn=entry,
                specs=build_gather_specs(*case),
                golden_fn=golden_gather,
                compare_fn={name: capture(name) for name in ("payload", "scales", "valid", "physical")},
                config=dict(backend_type="a5", execution_mode=False, device_id=args.device),
                save_data=True,
                compile_only=args.compile_only,
            )
            if not result.passed:
                raise SystemExit(result.error or 1)
            if args.compile_only:
                print(f"[编译产物] {result.work_dir}", flush=True)
                continue
            destination = work / "device-output" / "-".join(map(str, case)) / label
            destination.mkdir(parents=True, exist_ok=True)
            for name, value in captured.items():
                torch.save(value, destination / f"{name}.pt")
            if len(case) == 4 and case[-1] == "bytes":
                codes = torch.arange(256, dtype=torch.uint8)
                payload_codes = captured["payload"][:ROW_TILE].unique()
                scale_codes = captured["scales"].view(torch.uint8)[:, :ROW_TILE * SCALE_COUNT].unique()
                torch.testing.assert_close(payload_codes, codes, rtol=0, atol=0)
                torch.testing.assert_close(scale_codes, codes, rtol=0, atol=0)
                record["有效搬运覆盖全部256种编码"] = True
            outputs.append(captured)
            record[label] = str(result.work_dir)
            print(f"[产物] {result.work_dir}", flush=True)
        if args.compile_only:
            continue
        for name, actual in outputs[0].items():
            a = actual.contiguous().view(torch.uint8)
            b = outputs[1][name].contiguous().view(torch.uint8)
            torch.testing.assert_close(a, b, rtol=0, atol=0)
        record["不同字节数"] = 0
        records.append(record)
        print(f"[逐字节对照通过] {record}", flush=True)
        (work / "native-comparison.json").write_text(json.dumps(records, ensure_ascii=False, indent=2) + "\n")
