"""用当前 DSL 算子执行完整 decode forward，记录实际调度模式和输出冒烟检查。"""
import json
import os
from pathlib import Path
import runpy
import sys
import importlib

import golden
import torch
from pypto.pypto_core import ir, passes

work = Path(os.environ["DSV41_CAPTURE_DIR"]).resolve()
work.mkdir(parents=True, exist_ok=True)
compile_only = '--compile-only' in sys.argv
root = Path(os.environ['PYPTO_LIB_ROOT'])
entry = root / 'models/deepseek_v4_1_flash/decode_fwd.py'
sys.path.insert(0, str(entry.parent))
record_path = Path('<workspace>/artifacts/dsv41-hbg-forward-20261107/forward-compile.json')
original_run = golden.run

# 使用公开配置开启采集；只将诊断输出定向至当前采集目录。
from dataclasses import replace
from pypto.runtime import distributed_runner, runner
level = int(os.environ['DSV41_SWIMLANE_LEVEL'])
assert level in (1, 3)
normalizer = golden.normalize_config
def normalize_profile(config):
    return replace(normalizer(config), enable_chip_swimlane=level)
golden.normalize_config = normalize_profile
distributed_runner._run_directory = lambda compiled: work
def collect_profile(compiled):
    distributed_runner._collect_l3_swimlane(compiled.output_dir, compiled.platform, run_directory=work)
distributed_runner._collect_program_swimlane = collect_profile
conversions = []
def defer_conversion(work_dir, swimlane_dir, perf_file, func_names=None, deps_json=None):
    if perf_file is None:
        return
    base = work / 'dfx_outputs'
    conversions.append({
        'records': str(Path(perf_file).relative_to(base)),
        'names': str(Path(func_names).relative_to(base)) if func_names else None,
        'deps': str(Path(deps_json).relative_to(base)) if deps_json else None,
    })
    (work / 'conversions.json').write_text(json.dumps(conversions, ensure_ascii=False, indent=2) + '\n')
    print('转换任务已保存', conversions[-1]['records'], flush=True)
runner._generate_swimlane = defer_conversion


def compile_forward(**kwargs):
    if not compile_only:
        from types import SimpleNamespace
        record = json.loads(record_path.read_text())
        return SimpleNamespace(passed=True, work_dir=record['work_dir'])
    result = original_run(**kwargs)
    record_path.write_text(json.dumps({'passed': bool(result.passed), 'error': str(result.error or ''),
                                      'work_dir': str(result.work_dir), 'runtime': 'host_build_graph'},
                                     ensure_ascii=False, indent=2) + '\n')
    return result


def install_scheduler_probe():
    import simpler.worker as worker_module
    directory = work / 'scheduler'
    directory.mkdir(exist_ok=True)
    for previous in directory.glob('*.json'):
        previous.unlink()

    def observe(chip):
        record = {'进程': os.getpid(), 'mode': chip.scheduler_runtime_mode,
                  'publication': chip.scheduler_mode_publish_count}
        if record['publication']:
            (directory / f'{os.getpid()}.json').write_text(json.dumps(record) + '\n')
            print('SCHEDULER_PROBE', json.dumps(record), flush=True)

    original_child = worker_module._finalize_chip_worker_and_publish
    def child(buf, chip):
        try:
            observe(chip)
        finally:
            original_child(buf, chip)
    worker_module._finalize_chip_worker_and_publish = child

    original_teardown = worker_module.Worker._teardown_worker_tree
    def teardown(self, **kwargs):
        try:
            if self.level == 2 and self._chip_worker is not None:
                observe(self._chip_worker)
        finally:
            original_teardown(self, **kwargs)
    worker_module.Worker._teardown_worker_tree = teardown


golden.run = compile_forward
if not compile_only:
    assert os.environ.get('TASK_DEVICE'), '设备运行必须通过 task-submit'
    assert (ir.comm_layout.RANK_ID_OFFSET, ir.comm_layout.WINDOWS_IN_OFFSET) == (0, 16)
    install_scheduler_probe()
sys.argv[0] = str(entry)
with passes.PassContext([], runtime=passes.RuntimeKind.HOST_BUILD_GRAPH):
    namespace = runpy.run_path(str(entry), run_name='__main__')
if not compile_only:
    outputs = namespace['host_outputs']
    for name, tensor in outputs.items():
        assert torch.isfinite(tensor).all(), f'{name} 包含非有限值'
        assert torch.count_nonzero(tensor).item() == 0, f'零权重冒烟的 {name} 应全零'
        torch.save(tensor, work / (name + '.pt'))
    modes = [json.loads(p.read_text()) for p in (work / 'scheduler').glob('*.json')]
    assert len(modes) >= 2 and all(m['mode'] in (1, 2) and m['publication'] > 0 for m in modes), modes
    (work / 'forward-result.json').write_text(json.dumps({
        '通过': True, '采集级别': level, '层数': 40, '请求数': 12, '有效行数': 72,
        '专家数每卡': namespace['N_LOCAL_EXPERTS'], '设备': os.environ['TASK_DEVICE'],
        'scheduler': modes, '数值范围': '零权重冒烟；未验证真实权重整网精度',
    }, ensure_ascii=False, indent=2) + '\n')
    print(f'[FWD HBG L{level}] 40 层完成，AICore 异步调度已确认，零权重输出检查通过。', flush=True)
