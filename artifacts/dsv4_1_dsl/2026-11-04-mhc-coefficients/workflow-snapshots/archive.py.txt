"""Archive mHC RMS, HF32 projection and Sinkhorn coefficient validation."""

import gzip
import json
from pathlib import Path
import re


workspace = Path(__file__).resolve().parents[2]
work = workspace / "artifacts/dsv41-mhc-coeff-dsl"
repo = workspace / "pypto-a5-dev/pypto-lib-dsv41-dsl"
dest = repo / "artifacts/dsv4_1_dsl/2026-11-04-mhc-coefficients"
dest.mkdir(parents=True, exist_ok=True)
records = []


def normalize(value):
    value = value.replace(str(workspace), "${WORKSPACE}")
    value = value.replace(str(workspace.parent), "${USER_ROOT}")
    return "\n".join(line.rstrip() for line in value.splitlines()).rstrip() + "\n"


def copy(source, target, *, compressed=False, binary=False):
    raw = source.read_bytes()
    output = gzip.compress(raw, compresslevel=6, mtime=0) if compressed else raw
    if not compressed and not binary:
        output = normalize(raw.decode()).encode()
    target = dest / target
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(output)
    records.append({
        "source": str(source.relative_to(workspace)), "path": str(target.relative_to(dest)),
        "source_bytes": len(raw),
        "encoding": "gzip" if compressed else ("binary" if binary else "normalized UTF-8"),
    })


for source in sorted(work.glob("*.log")):
    if source.name != "archive-result.log":
        copy(source, Path("validation") / (source.name + ".txt"))
for source in sorted([*work.glob("*.py"), *work.glob("*.sh")]):
    copy(source, Path("workflow-snapshots") / (source.name + ".txt"))
for name in ("progress.txt", "validation-summary.json"):
    copy(work / name, name)

builds = {
    "pre-1": "_jit_l2_hc_pre_20261104_092429_3laf67bg",
    "sinkhorn-1": "_jit_l2_hc_sinkhorn_20261104_092438_ldds2xot",

    "sinkhorn-17": "_jit_l2_hc_sinkhorn_20261104_092152_fdzbn5_l",
    "project-17": "_jit_l2_hc_project_20261104_092326_3t_swvzq",
    "pre-72": "_jit_l2_hc_pre_20261104_092457_dgm3i6s8",

    "composed-swa": "_jit_l3_decode_swa_20261104_092435_f602d3rr/next_levels/l2_decode_swa",
}
for label, build_name in builds.items():
    build = repo / "build_output" / build_name
    if label != "composed-swa":
        for source in sorted((build / "data").rglob("*.pt")):
            target = Path("fixtures") / label / (str(source.relative_to(build / "data")) + ".gz")
            copy(source, target, compressed=True)
    for name in ("kernel_config.py", "compiled_meta.json"):
        if (build / name).exists():
            copy(build / name, Path("generated") / label / (name + ".txt"))
    for source in sorted((build / "orchestration").glob("*.cpp")):
        copy(source, Path("generated") / label / source.name)
    for source in sorted((build / "cache").glob("incore_*.bin")):
        copy(source, Path("generated") / label / source.name, binary=True)
    for source in sorted((build / "kernels").rglob("*.cpp")):
        copy(source, Path("generated") / label / source.relative_to(build))
    config = (build / "kernel_config.py").read_text()
    for name, entry in re.findall(r'"name": "([^"]+)", "source": \'([^\']+/entry.cpp)\'', config):
        for filename in ("entry.cpp", "device_callable.hpp", "manifest.json"):
            copy(Path(entry).with_name(filename), Path("generated") / label / name / filename)

for name in ("hc.py", "hc_dsl.py"):
    copy(repo / "models/deepseek_v4_1_flash" / name, Path("model-snapshot") / (name + ".txt"))
