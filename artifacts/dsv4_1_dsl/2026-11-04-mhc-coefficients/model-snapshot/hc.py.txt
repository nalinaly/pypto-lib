# model config
D = M.hidden_size
HC_MULT = M.hc_mult
HC_DIM = M.hc_dim
MIX_HC = M.mix_hc
HC_SINKHORN_ITER = M.hc_sinkhorn_iters
HC_EPS = M.hc_eps
NORM_EPS = M.rms_norm_eps

# tiling
MIX_PAD = 32
HC_PAD = 8
LINEAR_T_TILE = 16
LINEAR_K_TILE = 256
LINEAR_OK = 4
LINEAR_K_PER_SPLIT = HC_DIM // LINEAR_OK
PRE_T_TILE = 2
HEAD_D_TILE = 5120

# validation storage dimensions
RMS_ROWS_DYN = pl.dynamic("HC_RMS_ROWS")
PARTIAL_ROWS_DYN = pl.dynamic("HC_PARTIAL_ROWS")


@pl.jit.inline
def hc_sinkhorn(
    inv_rms: pl.Tensor, partials: pl.Tensor, scale: pl.Tensor, base: pl.Tensor,
    pre_mix: pl.Tensor, post_mix: pl.Tensor, residual_mix: pl.Tensor,
):
    """Schedule row-wise FP32 coefficient generation and return its joined TaskId."""
    tokens = pl.tensor.dim(pre_mix, 0)
    padded_rows = pl.tensor.dim(inv_rms, 0)
    residual_flat = pl.reshape(residual_mix, [tokens, HC_MULT * HC_MULT])
    ready = pl.system.task_dummy(deps=[])
    completed = pl.array.create(TASK_TILE, pl.TASK_ID)
    for i in pl.range(TASK_TILE):
        completed[i] = ready
    for row in pl.parallel(tokens):
        p0 = pl.slice(partials, [1, MIX_PAD], [row, 0])
        p1 = pl.slice(partials, [1, MIX_PAD], [padded_rows + row, 0])
        p2 = pl.slice(partials, [1, MIX_PAD], [2 * padded_rows + row, 0])
        p3 = pl.slice(partials, [1, MIX_PAD], [3 * padded_rows + row, 0])
        inverse = pl.slice(inv_rms, [1, 1], [row, 0])
        pre = pl.slice(pre_mix, [1, HC_MULT], [row, 0])
        post = pl.slice(post_mix, [1, HC_MULT], [row, 0])
        residual = pl.slice(residual_flat, [1, HC_MULT * HC_MULT], [row, 0])
        previous = completed[row % TASK_TILE]
        with pl.spmd(1, name_hint="dsl_mhc_sinkhorn", deps=[previous]) as done:
            coefficients_row(p0, p1, p2, p3, inverse, scale, base, pre, post, residual)
        completed[row % TASK_TILE] = done
    joined = pl.system.task_dummy(deps=[completed])
    return joined


@pl.jit.inline
def hc_project_into(x_hc: pl.Tensor, function: pl.Tensor, inv_rms: pl.Tensor, mixes_partials: pl.Tensor,
                    t_dim: pl.Scalar[pl.INDEX], hf32: pl.constexpr):
    """Compute original-stream RMS and four projection partials in supplied storage."""
    t_linear = pl.tensor.dim(inv_rms, 0)
    tokens = pl.tensor.dim(x_hc, 0)
    x_flat = pl.reshape(x_hc, [tokens, HC_DIM])
    for row in pl.parallel(t_dim):
        rms_source = pl.slice(x_flat, [1, HC_DIM], [row, 0])
        inverse = pl.slice(inv_rms, [1, 1], [row, 0])
        with pl.spmd(1, name_hint="dsl_mhc_rms"):
            inverse_rms_row(rms_source, inverse)

    if hf32:
        row_tiles = (t_dim + PROJECT_ROW_TILE - 1) // PROJECT_ROW_TILE
        for row_tile in pl.parallel(row_tiles):
            t0 = row_tile * PROJECT_ROW_TILE
            rows = pl.min(PROJECT_ROW_TILE, t_dim - t0)
            for split in pl.parallel(LINEAR_OK):
                k0 = split * LINEAR_K_PER_SPLIT
                project_source = pl.slice(x_flat, [PROJECT_ROW_TILE, LINEAR_K_PER_SPLIT], [t0, k0], valid_shape=[rows, LINEAR_K_PER_SPLIT])
                weight = pl.slice(function, [MIX_HC, LINEAR_K_PER_SPLIT], [0, k0])
                partial = split * t_linear + t0
                target = pl.slice(mixes_partials, [PROJECT_ROW_TILE, MIX_PAD], [partial, 0], valid_shape=[rows, MIX_HC])
                with pl.spmd(1, name_hint="dsl_mhc_hf32"):
                    project_hf32_tile(project_source, weight, target, t_dim, row_tile)
    else:
        for task in pl.spmd((t_linear // LINEAR_T_TILE) * LINEAR_OK, name_hint="mhc_linear"):
            t0 = (task // LINEAR_OK) * LINEAR_T_TILE
            split = task % LINEAR_OK
            k_base = split * LINEAR_K_PER_SPLIT
            valid_rows = pl.min(LINEAR_T_TILE, t_dim - t0)
            acc = pl.create_tensor([LINEAR_T_TILE, MIX_PAD], dtype=pl.FP32)
            for kb in pl.pipeline(LINEAR_K_PER_SPLIT // LINEAR_K_TILE, stage=2):
                k0 = k_base + kb * LINEAR_K_TILE
                linear_x_tile = pl.slice(
                    x_flat,
                    [LINEAR_T_TILE, LINEAR_K_TILE],
                    [t0, k0],
                    valid_shape=[valid_rows, LINEAR_K_TILE],
                )
                w_tile = pl.slice(
                    function,
                    [MIX_PAD, LINEAR_K_TILE],
                    [0, k0],
                    valid_shape=[MIX_HC, LINEAR_K_TILE],
                )
                acc = pl.matmul_acc(acc, linear_x_tile, w_tile, b_trans=True, init_cond=(kb == 0))
            partial = split * t_linear + t0
            mixes_partials[partial : partial + LINEAR_T_TILE, 0:MIX_PAD] = acc

    return inv_rms, mixes_partials


@pl.jit.inline
def hc_project(x_hc: pl.Tensor, function: pl.Tensor, hf32: pl.constexpr):
    """Allocate padded FP32 RMS and split-K storage for shared mHC projection."""
    tokens = pl.tensor.dim(x_hc, 0)
    padded_rows = ((tokens + LINEAR_T_TILE - 1) // LINEAR_T_TILE) * LINEAR_T_TILE
    inv_rms = pl.create_tensor([padded_rows, 1], dtype=pl.FP32)
    partials = pl.create_tensor([LINEAR_OK * padded_rows, MIX_PAD], dtype=pl.FP32)
    hc_project_into(x_hc, function, inv_rms, partials, tokens, hf32=hf32)
    return inv_rms, partials


@pl.jit.inline
def hc_head(
    x_hc: pl.Tensor,
    pre_mix: pl.Tensor,
    output: pl.Tensor,
):
    """Collapse HC streams with the delayed pre-mix and BF16 rounding."""
    t_dim = pl.tensor.dim(x_hc, 0)
    x_flat = pl.reshape(x_hc, [t_dim, HC_DIM])
    for block in pl.spmd((t_dim + PRE_T_TILE - 1) // PRE_T_TILE, name_hint="hc_collapse", allow_early_resolve=True):
        for t in pl.range(block * PRE_T_TILE, pl.min((block + 1) * PRE_T_TILE, t_dim)):
            pre_tile = pl.load(pre_mix, [t, 0], [1, HC_PAD], valid_shape=[1, HC_MULT])
            pre0 = pl.tile.read(pre_tile, [0, 0])
            pre1 = pl.tile.read(pre_tile, [0, 1])
            pre2 = pl.tile.read(pre_tile, [0, 2])
            pre3 = pl.tile.read(pre_tile, [0, 3])
            for d0 in pl.pipeline(0, D, HEAD_D_TILE, stage=2):
                x0 = pl.load(x_flat, [t, d0], [1, HEAD_D_TILE])
                x1 = pl.load(x_flat, [t, D + d0], [1, HEAD_D_TILE])
                x2 = pl.load(x_flat, [t, 2 * D + d0], [1, HEAD_D_TILE])
                x3 = pl.load(x_flat, [t, 3 * D + d0], [1, HEAD_D_TILE])
                y0 = pl.mul(x0, pre0)
                y1 = pl.mul(x1, pre1)
                y2 = pl.mul(x2, pre2)
                y3 = pl.mul(x3, pre3)
                y01 = pl.add(y0, y1)
                y23 = pl.add(y2, y3)
                y_tile = pl.add(y01, y23)
                y_bf16 = pl.cast(y_tile, target_type=pl.BF16, mode="rint")
                pl.store(y_bf16, [t, d0], output)
    return output


@pl.jit.inline
def hc_head_norm(
    x_hc: pl.Tensor,
    pre_mix: pl.Tensor,
    norm_weight: pl.Tensor[[D], pl.BF16],
    output: pl.Tensor,
):
    """Schedule one DSL collapse/RMSNorm task per row and return their joined TaskId."""
    tokens = pl.tensor.dim(x_hc, 0)
    x_flat = pl.reshape(x_hc, [tokens, HC_DIM])
    ready = pl.system.task_dummy(deps=[])
    completed = pl.array.create(TASK_TILE, pl.TASK_ID)
    for i in pl.range(TASK_TILE):
        completed[i] = ready
    for row in pl.parallel(tokens):
        source = pl.slice(x_flat, [1, HC_DIM], [row, 0])
        pre = pl.slice(pre_mix, [1, HC_MULT], [row, 0])
        target = pl.slice(output, [1, D], [row, 0])
        previous = completed[row % TASK_TILE]
        with pl.spmd(1, name_hint="dsl_hc_collapse_norm", deps=[previous]) as done:
            collapse_norm_row(source, pre, norm_weight, target)
        completed[row % TASK_TILE] = done
    joined = pl.system.task_dummy(deps=[completed])
    return joined


@pl.jit.inline
def hc_pre(
    x_hc: pl.Tensor,
    incoming_pre_mix: pl.Tensor,
    function: pl.Tensor,
    scale: pl.Tensor,
    base: pl.Tensor,
    output: pl.Tensor,
    next_pre_mix: pl.Tensor,
    post_mix: pl.Tensor,
    residual_mix: pl.Tensor,
    hf32: pl.constexpr,
):
    """Generate delayed HC coefficients and collapse the incoming streams."""
    inv_rms, partials = hc_project(x_hc, function, hf32=hf32)
    mixes_ready = hc_sinkhorn(inv_rms, partials, scale, base, next_pre_mix, post_mix, residual_mix)
    hc_head(x_hc, incoming_pre_mix, output)
    return mixes_ready


@pl.jit.inline
def zero_bf16_padding(x: pl.Tensor, valid_rows: pl.Scalar[pl.INT32]):
    """Clear physical rows that belong to an empty or partial TP slab.

    A TP rank with no logical token still publishes one fixed-capacity slab so
    every rank executes the same distributed window protocol.  Clearing the
    unused rows makes that padding deterministic and prevents an empty owner
    from publishing uninitialized data.
    """
    tokens = pl.tensor.dim(x, 0)
    if tokens > 0 and valid_rows < tokens:
        for row in pl.spmd(tokens, name_hint="decode_sp_zero_padding"):
            if row >= valid_rows:
                x[row : row + 1, 0:D] = pl.full([1, D], dtype=pl.BF16, value=0.0)


@pl.jit.inline
def hc_post(
    sublayer: pl.Tensor,
    residual: pl.Tensor,
    post_mix: pl.Tensor,
    residual_mix: pl.Tensor,
    output: pl.Tensor,
    deps: pl.Array[2, pl.TASK_ID],
):
    """Schedule DSL residual expansion after the supplied producer dependencies."""
    tokens = pl.tensor.dim(sublayer, 0)
    residual_flat = pl.reshape(residual, [tokens, HC_DIM])
    mix_flat = pl.reshape(residual_mix, [tokens, HC_MULT * HC_MULT])
    output_flat = pl.reshape(output, [tokens, HC_DIM])
    ready = pl.system.task_dummy(deps=[deps])
    completed = pl.array.create(TASK_TILE, pl.TASK_ID)
    for i in pl.range(TASK_TILE):
        completed[i] = ready
    for row in pl.parallel(tokens):
        hidden = pl.slice(sublayer, [1, D], [row, 0])
        skip = pl.slice(residual_flat, [1, HC_DIM], [row, 0])
        post = pl.slice(post_mix, [1, HC_MULT], [row, 0])
        mix = pl.slice(mix_flat, [1, HC_MULT * HC_MULT], [row, 0])
        target = pl.slice(output_flat, [1, HC_DIM], [row, 0])
        previous = completed[row % TASK_TILE]
        with pl.spmd(1, name_hint="dsl_hc_post", deps=[previous]) as done:
            expand_row(hidden, skip, post, mix, target)
        completed[row % TASK_TILE] = done
    return output


@pl.jit.inline
def prefill_sp_post(
    sublayer: pl.Tensor[[SP_T_DYN, D], pl.BF16],
    residual: pl.Tensor[[SP_T_DYN, HC_MULT, D], pl.FP32],
    post_mix: pl.Tensor[[SP_T_DYN, HC_MULT], pl.FP32],
    residual_mix: pl.Tensor[[SP_T_DYN, HC_MULT, HC_MULT], pl.FP32],
    output: pl.Tensor[[SP_T_DYN, HC_MULT, D], pl.FP32],
    next_pre_mix: pl.Tensor[[SP_T_DYN, HC_MULT], pl.FP32],
    tp_rank: pl.Scalar[pl.INT32],
    num_tokens: pl.Scalar[pl.INT32],
):
    """Run local mHC-post, writing active values and zero padding in one producer.

    A later pure-output clear is not an ordering edge for the post task. Each
    residual element is instead written exactly once, with hc_post's FP32
    accumulation order and FP32 output.
    """
    tokens = pl.tensor.dim(residual, 0)
    residual_flat = pl.reshape(residual, [tokens, HC_MULT * D])
    mix_flat = pl.reshape(residual_mix, [tokens, HC_MULT * HC_MULT])
    output_flat = pl.reshape(output, [tokens, HC_MULT * D])
    for block in pl.spmd(tokens * HC_MULT, name_hint="sp_mhc_post"):
        row = block // HC_MULT
        out_h = block % HC_MULT
        for col in pl.pipeline(0, D, 256, stage=2):
            value = pl.full([1, 256], dtype=pl.FP32, value=0.0)
            if tp_rank * tokens + row < num_tokens:
                hidden = pl.cast(sublayer[row:row + 1, col:col + 256], pl.FP32)
                value = pl.mul(hidden, pl.read(post_mix, [row, out_h]))
                for in_h in pl.unroll(HC_MULT):
                    lane = residual_flat[row:row + 1, in_h * D + col:in_h * D + col + 256]
                    value = pl.add(value, pl.mul(lane, pl.read(mix_flat, [row, in_h * HC_MULT + out_h])))
            output_flat[row:row + 1, out_h * D + col:out_h * D + col + 256] = value
    for row in pl.spmd(tokens, name_hint="sp_mix_zero_padding"):
        if tp_rank * tokens + row >= num_tokens:
            zero_mix = pl.full([1, 8], dtype=pl.FP32, value=0.0)
            next_pre_mix[row:row + 1, :] = pl.set_validshape(zero_mix, 1, HC_MULT)
    return output, next_pre_mix


def _round_hf32(value: torch.Tensor) -> torch.Tensor:
    """Round finite FP32 values to E8M10, with halfway values away from zero."""
    bits = value.contiguous().view(torch.int32)
    rounded = ((bits + 4096) & -8192).view(torch.float32)
    return torch.where(torch.isfinite(value), rounded, value)


def golden_mhc_mixes(
    x_hc: torch.Tensor,
    function: torch.Tensor,
    scale: torch.Tensor,
    base: torch.Tensor,
    iterations: int = M.hc_sinkhorn_iters,
    hf32: bool = False,
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """Project one HC stream into pre, post, and doubly-stochastic residual mixes."""
    flat = x_hc.flatten(-2).float()
    inverse_rms = torch.rsqrt(flat.square().mean(dim=-1, keepdim=True) + M.rms_norm_eps)
    projection_input = _round_hf32(flat) if hf32 else flat
    projection_weight = _round_hf32(function.float()) if hf32 else function.float()
    mixes = F.linear(projection_input, projection_weight) * inverse_rms
    return golden_hc_coefficients(mixes, scale, base, iterations)


def golden_hc_coefficients(mixes, scale, base, iterations=M.hc_sinkhorn_iters):
    """Generate FP32 delayed pre/post coefficients and the residual matrix."""
    hc = M.hc_mult
    pre_raw, post_raw, residual_raw = mixes.split((hc, hc, hc * hc), dim=-1)
    pre = torch.sigmoid(pre_raw * scale[0] + base[:hc]) + M.hc_eps
    post = 2 * torch.sigmoid(post_raw * scale[1] + base[hc : 2 * hc])
    residual_base = base[2 * hc :].unflatten(-1, (hc, hc))
    residual = (residual_raw.unflatten(-1, (hc, hc)) * scale[2] + residual_base).softmax(dim=-1)
    residual = residual + M.hc_eps
    residual = residual / (residual.sum(dim=-2, keepdim=True) + M.hc_eps)
    for _ in range(iterations - 1):
        residual = residual / (residual.sum(dim=-1, keepdim=True) + M.hc_eps)
        residual = residual / (residual.sum(dim=-2, keepdim=True) + M.hc_eps)
    return pre, post, residual


def golden_hc_collapse(x: torch.Tensor, pre_mix: torch.Tensor) -> torch.Tensor:
    """Collapse four HC streams with the kernel's pairwise FP32 reduction."""
    products = x.float() * pre_mix.float().unsqueeze(-1)
    first_pair = products[..., 0, :] + products[..., 1, :]
    second_pair = products[..., 2, :] + products[..., 3, :]
    return (first_pair + second_pair).to(x.dtype)


def golden_hc_post(
    sublayer: torch.Tensor,
    residual: torch.Tensor,
    post_mix: torch.Tensor,
    residual_mix: torch.Tensor,
) -> torch.Tensor:
    """Expand a sublayer result and mix the four residual streams in FP32."""
    update = post_mix.unsqueeze(-1) * sublayer.unsqueeze(-2)
    skip = (residual_mix.unsqueeze(-1) * residual.unsqueeze(-2)).sum(dim=-3)
    return update.float() + skip.float()


def golden_l2_hc_pre(tensors):
    """Compute the four outputs with the incoming mix independent of the next mix."""
    pre_mix, post_mix, residual_mix = golden_mhc_mixes(
        tensors["x_hc"], tensors["function"], tensors["scale"], tensors["base"],
        hf32=True,
    )
    tensors["pre_mix"][:] = pre_mix
    tensors["post_mix"][:] = post_mix
    tensors["residual_mix"][:] = residual_mix
    collapsed = golden_hc_collapse(tensors["x_hc"], tensors["incoming_pre_mix"])
    tensors["output"][:] = collapsed.to(torch.bfloat16)


def golden_l2_hc_post(tensors):
    """Fill the expanded residual stream with FP32 accumulation and storage."""
    tensors["output"][:] = golden_hc_post(
        tensors["sublayer"], tensors["residual"], tensors["post_mix"], tensors["residual_mix"],
    )


def golden_l2_hc_head(tensors):
    """Fill the expected collapsed stream output."""
    tensors["output"][:] = golden_hc_collapse(tensors["x_hc"], tensors["pre_mix"]).to(torch.bfloat16)


def golden_l2_hc_head_norm(tensors):
    """Normalize the BF16-rounded collapse with independent CPU RMS statistics."""
    collapsed = golden_hc_collapse(tensors["x_hc"], tensors["pre_mix"]).bfloat16().float()
    inverse = torch.rsqrt(collapsed.square().mean(dim=-1, keepdim=True) + NORM_EPS)
    normalized = collapsed * inverse * tensors["norm_weight"].float()
    tensors["output"].copy_(normalized.bfloat16())


def golden_l2_hc_sinkhorn(tensors):
    """Check coefficient generation independently of the projection and RMS kernels."""
    tokens = tensors["pre_mix"].shape[0]
    padded = tensors["inv_rms"].shape[0]
    parts = tensors["partials"].reshape(LINEAR_OK, padded, MIX_PAD)[:, :tokens, :MIX_HC]
    mixed = ((parts[0] + parts[1]) + parts[2]) + parts[3]
    mixed = mixed * tensors["inv_rms"][:tokens]
    pre, post, residual = golden_hc_coefficients(mixed, tensors["scale"], tensors["base"])
    tensors["pre_mix"].copy_(pre)
    tensors["post_mix"].copy_(post)
    tensors["residual_mix"].copy_(residual)


def golden_l2_hc_project(tensors, *, hf32=True):
    """Check original-stream RMS and each independently rounded split-K projection."""
    flat = tensors["x_hc"].flatten(1).float()
    tokens = flat.shape[0]
    padded = tensors["inv_rms"].shape[0]
    tensors["inv_rms"][:tokens].copy_(torch.rsqrt(flat.square().mean(dim=-1, keepdim=True) + NORM_EPS))
    weight = tensors["function"].float()
    lhs = _round_hf32(flat) if hf32 else flat
    rhs = _round_hf32(weight) if hf32 else weight
    partials = tensors["partials"].reshape(LINEAR_OK, padded, MIX_PAD)
    for split in range(LINEAR_OK):
        start = split * LINEAR_K_PER_SPLIT
        end = start + LINEAR_K_PER_SPLIT
        partials[split, :tokens, :MIX_HC].copy_(F.linear(lhs[:, start:end], rhs[:, start:end]))


def compare_hc_projection(actual, expected, *, inputs, check_padding=True, **kwargs):
    """Check projection numerics, exact midpoint probes, and untouched padded storage."""
    from golden import ratio_allclose

    tokens = inputs["x_hc"].shape[0]
    padded = actual.shape[0] // LINEAR_OK
    got = actual.reshape(LINEAR_OK, padded, MIX_PAD)
    ref = expected.reshape(LINEAR_OK, padded, MIX_PAD)
    compare = ratio_allclose(atol=2e-5, rtol=5e-4, max_error_ratio=0.0)
    passed, detail = compare(got[:, :tokens, :MIX_HC], ref[:, :tokens, :MIX_HC], inputs=inputs, **kwargs)
    if not passed:
        return passed, detail
    exact = ratio_allclose(atol=0.0, rtol=0.0, max_error_ratio=0.0)
    passed, detail = exact(got[:, :tokens, :2], ref[:, :tokens, :2], inputs=inputs, **kwargs)
    if not passed:
        return passed, "HF32 midpoint/FP32 probes: " + detail
    if not check_padding:
        return passed, detail
    care = torch.ones_like(ref, dtype=torch.bool)
    care[:, :tokens, :MIX_HC] = False
    return exact(got[care], ref[care], inputs=inputs, **kwargs)


def compare_hc_inverse(actual, expected, *, inputs, **kwargs):
    """Check FP32 inverse RMS and exact preservation of padded rows."""
    from golden import ratio_allclose

    tokens = inputs["x_hc"].shape[0]
    compare = ratio_allclose(atol=1e-6, rtol=2e-6, max_error_ratio=0.0)
    passed, detail = compare(actual[:tokens], expected[:tokens], inputs=inputs, **kwargs)
    if not passed:
        return passed, detail
    exact = ratio_allclose(atol=0.0, rtol=0.0, max_error_ratio=0.0)
    return exact(actual[tokens:], expected[tokens:], inputs=inputs, **kwargs)


def compare_hc_post_order(actual, expected, *, inputs, **kwargs):
    """Check both the model golden and exact sequential FP32 accumulation."""
    from golden import ratio_allclose

    compare = ratio_allclose(atol=1e-6, rtol=1e-5)
    passed, detail = compare(actual, expected, inputs=inputs, **kwargs)
    if not passed:
        return passed, detail
    ordered = inputs["sublayer"].float().unsqueeze(1) * inputs["post_mix"].unsqueeze(-1)
    for in_h in range(HC_MULT):
        residual = inputs["residual"][:, in_h:in_h + 1, :]
        factor = inputs["residual_mix"][:, in_h, :].unsqueeze(-1)
        ordered = ordered + residual * factor
    exact = ratio_allclose(atol=0.0, rtol=0.0, max_error_ratio=0.0)
    return exact(actual, ordered, inputs=inputs, **kwargs)


HC_INPUT_NAMES = ("x_hc", "pre_mix", "hc_attn_fn", "hc_attn_scale", "hc_attn_base", "attn_norm_weight")


def make_hc_inputs(tokens, seed, case):
    """mHC inputs with distinct per-token magnitudes so both RMS statistics stay observable."""
    gen = torch.Generator().manual_seed(seed)
    magnitudes = (0.5 + torch.arange(tokens) % 4).reshape(-1, 1, 1)
    x_hc = (torch.randn(tokens, HC_MULT, D, generator=gen) * magnitudes).bfloat16().float()
    values = {
        "x_hc": x_hc,
        "pre_mix": torch.sigmoid(torch.randn(tokens, HC_MULT, generator=gen)) + M.hc_eps,
        "hc_attn_fn": torch.randn(MIX_HC, HC_DIM, generator=gen) / math.sqrt(HC_DIM),
        "hc_attn_scale": torch.randn(3, generator=gen),
        "hc_attn_base": torch.randn(MIX_HC, generator=gen),
        "attn_norm_weight": (torch.randn(D, generator=gen) * 0.1 + 1).bfloat16(),
    }
    if case == "zero":
        values["x_hc"] = torch.zeros_like(x_hc)
    return values

def build_l2_hc_pre_specs(batch: int = 33, sequence: int = 1):

def build_l2_hc_post_specs(batch: int = 33, sequence: int = 1):
    """Build BF16-valued residual streams and nonsymmetric mixing coefficients."""
    from golden import TensorSpec

    tokens = batch * sequence
    generator = torch.Generator().manual_seed(7)

    def init_sublayer():
        return torch.randn(tokens, D, generator=generator).bfloat16()

    def init_residual():
        return torch.randn(tokens, HC_MULT, D, generator=generator)

    def init_post_mix():
        return 2 * torch.sigmoid(torch.randn(tokens, HC_MULT, generator=generator))

    def init_residual_mix():
        mix = torch.rand(tokens, HC_MULT, HC_MULT, generator=generator) + HC_EPS
        for _ in range(HC_SINKHORN_ITER):
            mix = mix / (mix.sum(dim=-1, keepdim=True) + HC_EPS)
            mix = mix / (mix.sum(dim=-2, keepdim=True) + HC_EPS)
        return mix

    return [
        TensorSpec("sublayer", [tokens, D], torch.bfloat16, init_value=init_sublayer),
        TensorSpec("residual", [tokens, HC_MULT, D], torch.float32, init_value=init_residual),
        TensorSpec("post_mix", [tokens, HC_MULT], torch.float32, init_value=init_post_mix),
        TensorSpec("residual_mix", [tokens, HC_MULT, HC_MULT], torch.float32, init_value=init_residual_mix),
        TensorSpec("output", [tokens, HC_MULT, D], torch.float32),
    ]

def build_l2_hc_head_norm_specs(batch: int = 33, sequence: int = 1):

def build_l2_hc_sinkhorn_specs(batch: int = 33, sequence: int = 1):
    """Build padded partials with cancellation, zero values and inactive NaNs."""
    from golden import TensorSpec

    tokens = batch * sequence
    padded = (tokens + LINEAR_T_TILE - 1) // LINEAR_T_TILE * LINEAR_T_TILE
    generator = torch.Generator().manual_seed(23)

    def init_inverse():
        value = torch.full((padded, 1), float("nan"))
        value[:tokens] = torch.rand(tokens, 1, generator=generator) + 0.5
        return value

    def init_partials():
        value = torch.full((LINEAR_OK, padded, MIX_PAD), float("nan"))
        value[:, :tokens, :MIX_HC] = torch.randn(LINEAR_OK, tokens, MIX_HC, generator=generator)
        value[:, 0, :MIX_HC] = 0.0
        if tokens > 1:
            value[:, 1, :MIX_HC] = torch.tensor([2**24, 1, 1, -(2**24)]).float().reshape(-1, 1)
        return value.reshape(LINEAR_OK * padded, MIX_PAD)

    def init_base():
        value = torch.randn(MIX_HC, generator=generator)
        value[:4] = torch.tensor([-60.0, 60.0, -1.0, 1.0])
        return value

    return [
        TensorSpec("inv_rms", [padded, 1], torch.float32, init_value=init_inverse),
        TensorSpec("partials", [LINEAR_OK * padded, MIX_PAD], torch.float32, init_value=init_partials),
        TensorSpec("scale", [3], torch.float32, init_value=torch.tensor([0.7, -0.3, 1.2])),
        TensorSpec("base", [MIX_HC], torch.float32, init_value=init_base),
        TensorSpec("pre_mix", [tokens, HC_MULT], torch.float32),
        TensorSpec("post_mix", [tokens, HC_MULT], torch.float32),
        TensorSpec("residual_mix", [tokens, HC_MULT, HC_MULT], torch.float32),
    ]

def build_l2_hc_project_specs(batch: int = 33, sequence: int = 1):
    """Build positive/negative HF32 ties for both operands and zero/near-zero RMS rows."""
    from golden import TensorSpec

    tokens = batch * sequence
    padded = (tokens + LINEAR_T_TILE - 1) // LINEAR_T_TILE * LINEAR_T_TILE
    generator = torch.Generator().manual_seed(29)

    def init_x_hc():
        value = torch.randn(tokens, HC_DIM, generator=generator)
        signs = torch.where(torch.arange(tokens) % 2 == 0, 1.0, -1.0)
        for split in range(LINEAR_OK):
            start = split * LINEAR_K_PER_SPLIT
            value[:, start] = signs * (1.0 + 2**-11)
            value[:, start + 1] = 1.0
        if tokens > 2:
            value[-1].zero_()
            value[-2].mul_(1e-7)
        return value.reshape(tokens, HC_MULT, D)

    def init_function():
        value = torch.randn(MIX_HC, HC_DIM, generator=generator) / math.sqrt(HC_DIM)
        value[:2].zero_()
        for split in range(LINEAR_OK):
            start = split * LINEAR_K_PER_SPLIT
            value[0, start] = 1.0
            value[1, start + 1] = (-1.0 if split % 2 else 1.0) * (1.0 + 2**-11)
        return value

    return [
        TensorSpec("x_hc", [tokens, HC_MULT, D], torch.float32, init_value=init_x_hc),
        TensorSpec("function", [MIX_HC, HC_DIM], torch.float32, init_value=init_function),
        TensorSpec("inv_rms", [padded, 1], torch.float32, init_value=-123.5),
        TensorSpec("partials", [LINEAR_OK * padded, MIX_PAD], torch.float32, init_value=-123.5),
    ]


if __name__ == "__main__":
    import argparse
    from functools import partial

    from golden import ratio_allclose, run

    parser = argparse.ArgumentParser(description="DeepSeek V4.1 HC coefficients, collapse, RMSNorm and post")
    parser.add_argument("-p", "--platform", dest="backend_type", choices=["a5"], default="a5")
    parser.add_argument("--sim", dest="execution_mode", action="store_true")
    parser.add_argument("-d", "--device", type=int, default=0)
    parser.add_argument("--batch", type=int, default=33)
    parser.add_argument("--sequence", type=int, default=1)
    operations = ["pre", "post", "head", "head-norm", "sinkhorn", "project", "boundary", "coefficients", "all"]
    parser.add_argument("--op", choices=operations, default="all")
    parser.add_argument("--full-fp32", action="store_true", help="Check the existing FP32 projection branch")
    parser.add_argument("--compile-only", action="store_true")
    parser.add_argument("--save-data", action="store_true")
    parser.add_argument("--golden-data")
    parser.add_argument("--runtime-dir")
    parser.add_argument("--enable-chip-swimlane", type=int, choices=[1, 2, 3, 4], default=0)
    args = parser.parse_args()
    if args.op in ("all", "boundary", "coefficients") and (args.golden_data or args.runtime_dir):
        parser.error("replay requires selecting one --op")
    cases = [
        ("pre", l2_hc_pre, build_l2_hc_pre_specs, golden_l2_hc_pre),
        ("post", l2_hc_post, build_l2_hc_post_specs, golden_l2_hc_post),
        ("head", l2_hc_head, build_l2_hc_head_specs, golden_l2_hc_head),
        ("head-norm", l2_hc_head_norm, build_l2_hc_head_norm_specs, golden_l2_hc_head_norm),
        ("sinkhorn", l2_hc_sinkhorn, build_l2_hc_sinkhorn_specs, golden_l2_hc_sinkhorn),
        ("project", l2_hc_project, build_l2_hc_project_specs, golden_l2_hc_project),
    ]
    failures = []
    for name, fn, build_specs, golden_fn in cases:
        selected = ("head-norm", "post") if args.op == "boundary" else (args.op,)
        if args.op == "coefficients":
            selected = ("sinkhorn", "project", "pre")
        if args.op != "all" and name not in selected:
            continue
        print(f"[HC] {name}", flush=True)
        if name == "pre":
            compare_fn = {
                "pre_mix": ratio_allclose(atol=2.5e-5, rtol=5e-3),
                "post_mix": ratio_allclose(atol=2.5e-5, rtol=5e-3),
                "residual_mix": ratio_allclose(atol=2.5e-5, rtol=5e-3),
            }
            compare_fn["output"] = ratio_allclose(atol=1e-4, rtol=1.0 / 128)
        elif name == "post":
            compare_fn = {"output": compare_hc_post_order}
        elif name == "head-norm":
            compare_fn = {"output": ratio_allclose(atol=1e-6, rtol=1.0 / 128, max_error_ratio=0.0)}
        elif name == "sinkhorn":
            compare = ratio_allclose(atol=2e-6, rtol=1e-5, max_error_ratio=0.0)
            compare_fn = {"pre_mix": compare, "post_mix": compare, "residual_mix": compare}
        elif name == "project":
            compare_fn = {"inv_rms": compare_hc_inverse, "partials": compare_hc_projection}
        else:
            compare_fn = {"output": ratio_allclose(atol=1e-4, rtol=1.0 / 128)}
        config = {"backend_type": args.backend_type, "execution_mode": args.execution_mode}
        config.update(device_id=args.device, enable_chip_swimlane=args.enable_chip_swimlane)
        if name == "project" and args.full_fp32:
            fn = l2_hc_project_fp32
            golden_fn = partial(golden_l2_hc_project, hf32=False)
            # The existing FP32 projection does not define its unused scratch columns.
            compare_fn["partials"] = partial(compare_hc_projection, check_padding=False)
        result = run(
            fn=fn,
            specs=build_specs(args.batch, args.sequence),
            golden_fn=golden_fn,
            config=config,
            rtol=1e-3,
            atol=1e-3,
            compare_fn=compare_fn,
            compile_only=args.compile_only,
            save_data=args.save_data,
            golden_data=args.golden_data,
            runtime_dir=args.runtime_dir,
        )
        if not result.passed:
            failures.append(f"{name}: {result.error or 'validation failed'}")
    if failures:
        raise SystemExit("; ".join(failures))
