inventory = []


def capture(source, target, binary=False):
    destination = archive / target
    destination.parent.mkdir(parents=True, exist_ok=True)
    if binary:
        with source.open("rb") as inp, gzip.open(destination, "wb") as out:
            shutil.copyfileobj(inp, out)
    else:
        value = source.read_text().replace(str(workspace), "<workspace>")
        value = value.replace(str(workspace.parent), "<user-home>")
        destination.write_text("\n".join(line.rstrip() for line in value.splitlines()).rstrip() + "\n")
    inventory.append({"path": str(destination.relative_to(archive)), "source": str(source.relative_to(workspace)),
                      "bytes": destination.stat().st_size, "encoding": "gzip" if binary else "utf-8"})


def generated(label, directory, fixtures=False):
    for name in ("kernel_config.py", "compiled_meta.json"):
        source = directory / name
        if source.is_file():
            capture(source, f"generated/{label}/{name}.txt")
    for source in sorted((directory / "orchestration").glob("*.cpp")):
        capture(source, f"generated/{label}/orchestration/{source.name}.txt")
    for source in sorted((directory / "kernels").rglob("*.cpp")):
        if "proj_cmp" in source.name or "unused_attention_output" in source.name:
            capture(source, f"generated/{label}/kernels/{source.name}.txt")
    config = directory / "kernel_config.py"
    if config.is_file():
        for name, entry in re.findall(r'"name": "([^"]+)", "source": \'([^\']+)\'', config.read_text()):
            if name.startswith("project_compressor_tile"):
                for filename in ("entry.cpp", "device_source.hpp", "manifest.json"):
                    capture(Path(entry).parent / filename, f"generated/{label}/foreign/{name}/{filename}.txt")
    if fixtures:
        for source in sorted((directory / "data").rglob("*.pt")):
            capture(source, Path("fixtures") / label / f"{source.relative_to(directory / 'data')}.gz", True)

review = []
for name in ("decode_c2a_full", "prefill_c2a_full", "decode_layer"):
    before = ast.parse((work / (name + "_before.py")).read_text())
    after = ast.parse((model / "models/deepseek_v4_1_flash" / (name + ".py")).read_text())
    def communication(tree):
        return [ast.dump(node) for node in ast.walk(tree) if isinstance(node, ast.Call)
                and isinstance(node.func, ast.Attribute) and ast.unparse(node.func).startswith("pld.")]
    if communication(before) != communication(after):
        raise RuntimeError("Communication changed: " + name)
    review.append(name + ": all distributed calls unchanged")

new = ast.parse((model / "models/deepseek_v4_1_flash/compressor.py").read_text())

b = {node.name: node for node in new.body if isinstance(node, ast.FunctionDef)}

new = ast.parse((model / "models/deepseek_v4_1_flash/decode_c2a_full.py").read_text())
run = next(n for n in ast.walk(old.body[-1]) if isinstance(n, ast.Call)
           and isinstance(n.func, ast.Name) and n.func.id == "run")
expected = next(k.value for k in run.keywords if k.arg == "compare_fn")
class Normalize(ast.NodeTransformer):
    def visit_Attribute(self, node):
        if isinstance(node.value, ast.Name) and node.value.id == "args" and node.attr == "tokens":
            return ast.Name(id="tokens", ctx=ast.Load())
        return self.generic_visit(node)
expected = Normalize().visit(expected)
factory = next(n for n in new.body if isinstance(n, ast.FunctionDef) and n.name == "c2a_comparators")
actual = next(n.value for n in factory.body if isinstance(n, ast.Return))
if ast.dump(expected) != ast.dump(actual):
    raise RuntimeError("C2A comparator expressions changed")
review.append("C2A comparator expressions and thresholds: unchanged")
old = ast.parse((work / "decode_layer_before.py").read_text())
new = ast.parse((model / "models/deepseek_v4_1_flash/decode_layer.py").read_text())
run = next(n for n in ast.walk(old.body[-1]) if isinstance(n, ast.Call)
           and isinstance(n.func, ast.Name) and n.func.id == "run")
expected = next(k.value for k in run.keywords if k.arg == "compare_fn")
actual = next(n.value for n in ast.walk(new.body[-1]) if isinstance(n, ast.Assign)
              and any(isinstance(target, ast.Name) and target.id == "comparators" for target in n.targets))
if ast.dump(expected) != ast.dump(actual):
    raise RuntimeError("SWA comparator expressions changed")
review.append("SWA comparator expressions and thresholds: unchanged")
