"""Capture send-staging validation, generated code and CPU fixtures."""

import ast
import gzip
import json
from pathlib import Path
import re
import shutil
import subprocess


workspace = Path(__file__).resolve().parents[2]
model = workspace / "pypto-a5-dev/pypto-lib-dsv41-dsl"
work = workspace / "artifacts/dsv41-dispatch-stage-dsl"
archive = model / "artifacts/dsv4_1_dsl/2026-11-05-dispatch-stage"
archive.mkdir(parents=True, exist_ok=True)
inventory = []


def capture(source, target, binary=False):
    destination = archive / target
    destination.parent.mkdir(parents=True, exist_ok=True)
    if binary:
        with source.open("rb") as inp, gzip.open(destination, "wb") as out:
            shutil.copyfileobj(inp, out)
    else:
        value = source.read_text().replace(str(workspace), "<workspace>")
        value = value.replace(str(workspace.parent), "<user-home>")
        destination.write_text("\n".join(line.rstrip() for line in value.splitlines()).rstrip() + "\n")
    inventory.append({"path": str(destination.relative_to(archive)),
                      "source": str(source.relative_to(workspace)),
                      "bytes": destination.stat().st_size, "encoding": "gzip" if binary else "utf-8"})


def generated(label, directory, fixtures=False):
    for name in ("kernel_config.py", "compiled_meta.json"):
        source = directory / name
        if source.is_file():
            capture(source, f"generated/{label}/{name}.txt")
    for source in sorted((directory / "orchestration").glob("*.cpp")):
        capture(source, f"generated/{label}/orchestration/{source.name}.txt")
    config = directory / "kernel_config.py"
    if config.is_file():
        for name, entry in re.findall(r'"name": "([^"]+)", "source": \'([^\']+)\'', config.read_text()):
            if name.startswith("stage_rows"):
                for filename in ("entry.cpp", "device_source.hpp", "manifest.json"):
                    capture(Path(entry).parent / filename, f"generated/{label}/foreign/{name}/{filename}.txt")
    if fixtures:
        for source in sorted((directory / "data").rglob("*.pt")):
            capture(source, Path("fixtures") / label / f"{source.relative_to(directory / 'data')}.gz", True)


local_log = (work / "local.log").read_text()
layer_log = (work / "layer0-ep2.log").read_text()
stage_cases = re.findall(r"^\[DISPATCH_STAGE_DSL\] rows=(\d+) active=(-?\d+)", local_log, re.MULTILINE)
gather_cases = re.findall(r"^\[DISPATCH_GATHER_DSL\] case=(\w+)", local_log, re.MULTILINE)
if len(stage_cases) != 10 or len(gather_cases) != 4:
    raise RuntimeError("Expected ten stage cases and four gather regressions")
if len(re.findall(r"^\[RUN\] PASS", local_log, re.MULTILINE)) != 14:
    raise RuntimeError("All local cases must pass before capture")
outputs = len(re.findall(r"^\[RUN\]   '[^']+' PASS", layer_log, re.MULTILINE))
if "[RUN] PASS" not in layer_log or outputs != 16:
    raise RuntimeError("The EP2 layer must pass all 16 comparisons before capture")

review = []
for name in ("decode_moe", "moe_dispatch_dsl"):
    old = ast.parse((work / (name + "_before.py")).read_text())
    new = ast.parse((model / "models/deepseek_v4_1_flash" / (name + ".py")).read_text())
    a = {node.name: node for node in old.body if isinstance(node, ast.FunctionDef)}
    b = {node.name: node for node in new.body if isinstance(node, ast.FunctionDef)}
    for key in a:
        if name == "decode_moe" and key == "dispatch":
            before = [ast.dump(node) for node in a[key].body]
            after = [ast.dump(node) for node in b[key].body]
            if before[:2] != after[:2] or before[3:] != after[3:]:
                raise RuntimeError("A statement outside the stage body changed")
            review.append("dispatch: input/output allocations and all statements after staging unchanged")
        elif ast.dump(a[key]) != ast.dump(b[key]):
            raise RuntimeError("Unrelated function changed: " + name + "." + key)
        else:
            review.append(name + "." + key + ": unchanged")
subprocess.run(["git", "diff", "--exit-code", "HEAD", "--", "models/deepseek_v4_1_flash/expert_routed.py",
                "models/deepseek_v4_1_flash/decode_layer.py", "models/deepseek_v4_1_flash/prefill_moe.py"],
               cwd=model, check=True)
review.extend(["expert_routed.py: unchanged, including ffn_reduce_scatter", "decode_layer.py: unchanged",
               "prefill_moe.py: unchanged, separate staging implementation retained"])
(work / "communication-review.txt").write_text("\n".join(review) + "\n")

for name in ("local.log", "layer0-ep2.log", "local-compile.log", "moe-compile.log"):
    capture(work / name, f"validation/{name}.txt")
capture(work / "communication-review.txt", "validation/communication-review.txt")
for name in ("decode_moe", "moe_dispatch_dsl"):
    capture(model / "models/deepseek_v4_1_flash" / (name + ".py"), f"model-snapshot/{name}.py.txt")
    capture(work / (name + "_before.py"), f"model-snapshot/{name}_before.py.txt")
generated("stage32", model / "build_output/_jit_validate_stage_20261105_100515_64htd0em", True)
generated("stage80", model / "build_output/_jit_validate_stage_20261105_100525_k_mf8twr", True)
layer = max((model / "build_output").glob("_jit_l3_decode_layer_*"), key=lambda path: path.stat().st_mtime)
generated("layer0-ep2", layer)
generated("layer0-ep2/chip", layer / "next_levels/l2_decode_layer")
capture(Path(__file__).resolve(), "workflow/archive.py.txt")


def revision(path):
    return subprocess.check_output(["git", "-C", str(path), "rev-parse", "HEAD"], text=True).strip()


summary = {
    "status": "PASS", "branch": "dsv4.1-dsl", "model_base": revision(model),
    "pypto_revision": revision(workspace / "issue75-env/pypto-pr"),
    "dsl_revision": revision(workspace / "cannbot-dsl-dev"),
    "replaced_task": "dispatch_stage", "replacement_task": "dsl_dispatch_stage",
    "communication_code_unchanged": True, "gather_functions_unchanged": True,
    "stage_cases": [{"rows": int(rows), "active": int(active), "result": "PASS"} for rows, active in stage_cases],
    "gather_regressions": gather_cases,
    "local_acceptance": "Byte-exact outputs, including inactive scale rows and output guards",
    "local_task": re.search(r"task_\d+_\d+_\d+", local_log).group(0),
    "ep2_task": re.search(r"task_\d+_\d+_\d+", layer_log).group(0),
    "ep2_devices": [int(value) for value in re.findall(r"已获取设备 (\d+) 的锁", layer_log)],
    "layer_outputs": outputs,
    "limits": ["A5 generated CPU fixtures", "No checkpoint/prefill or performance claim",
               "Original layer comparators retained"],
}
(archive / "validation-summary.json").write_text(json.dumps(summary, indent=2) + "\n")
(archive / "artifact-manifest.json").write_text(json.dumps(inventory, indent=2) + "\n")
print(f"Captured {len(inventory)} files; {sum(item['bytes'] for item in inventory)} bytes")
