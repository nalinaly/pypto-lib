def category(name):
    if "window_kv_page" in name or "norm_rope_publish" in name:
        return "kv"
    if "_gather_sparse_vf" in name:
        return "attention"
    if "_proj_q_" in name or "_proj_kv" in name:
        return "qkv"
    if "q_norm_quant" in name or "qkv_input_quant" in name:
        return "norm"
    if "_o_proj" in name or "_rope_quant_o" in name or "quant_o_a" in name:
        return "output"
    if "zero_padding" in name:
        return "padding"
    return "hc"


def load_variant(key, title):
    meta = json.loads((ROOT / f"{key}.json").read_text())
    if not meta["passed"]:
        raise ValueError(f"Cannot present unvalidated capture: {key}")
    directory = Path(meta["work_dir"]) / "dfx_outputs/rank0/d0"
    raw = json.loads((directory / "chip_swimlane_records.json").read_text())
    if raw["chip_swimlane_level"] != 1:
        raise ValueError("Expected L1 AICore timing")
    merged_path, = directory.glob("merged_swimlane_*.json")
    trace = json.loads(merged_path.read_text())
    names = {(e.get("pid"), e.get("tid")): e["args"]["name"]
             for e in trace["traceEvents"] if e.get("ph") == "M" and e["name"] == "thread_name"}
    raw_events = [e for e in trace["traceEvents"] if e.get("ph") == "X"]
    if len(raw_events) != len(raw["aicore_tasks"]):
        raise ValueError("Incomplete L1 conversion")
    zero = min(e["ts"] for e in raw_events)
    events = []
    for e in raw_events:
        lane = names[(e["pid"], e["tid"])]
        if not lane.startswith(("AIC_", "AIV_")) or e["dur"] < 0:
            raise ValueError(f"Invalid AICore event: {e}")
        family = e["name"].split("(")[0]
        events.append(dict(lane=lane, start=e["ts"] - zero, duration=e["dur"],
                           name=e["name"], family=family, category=category(family),
                           task=str(e["args"]["taskId"]), hint=e["args"].get("event-hint", "")))
    lanes = sorted(set(names.values()), key=lambda x: (x.startswith("AIV"), int(x.split("_")[1])))
    by_lane = defaultdict(list)
    for event in events:
        by_lane[event["lane"]].append(event)
    for lane, values in by_lane.items():
        end = -math.inf
        for event in sorted(values, key=lambda x: x["start"]):
            if event["start"] < end - 0.001:
                raise ValueError(f"Overlapping physical-core records on {lane}")
            end = event["start"] + event["duration"]
    kv = [e for e in events if e["category"] == "kv"]
    observed = max(e["start"] + e["duration"] for e in events)
    kv_start = min(e["start"] for e in kv)
    kv_end = max(e["start"] + e["duration"] for e in kv)
    summary = dict(key=key, title=title, device=meta["device"], passed=True, level=1,
                   lanes=lanes, events=events, observed_us=observed,
                   execution_count=len(events), task_count=len({e["task"] for e in events}),
                   kv_executions=len(kv), kv_tasks=len({e["task"] for e in kv}),
                   kv_start_us=kv_start, kv_end_us=kv_end, kv_span_us=kv_end-kv_start,
                   families=dict(Counter(e["family"] for e in events)), codec=meta["cache_codec"])
    raw_out = ROOT / "raw" / key
    raw_out.mkdir(parents=True, exist_ok=True)
    for filename in ("chip_swimlane_records.json", "deps.json", "name_map.json"):
        shutil.copy2(directory / filename, raw_out / filename)
    shutil.copy2(merged_path, ROOT / f"{key}-l1.perfetto.json")
    return summary, meta

def plot_lanes(ax, variant, limits, focus=False, compact=False):
    lanes = [x for x in variant["lanes"] if not focus or x.startswith("AIV_")]
    positions = {lane: i for i, lane in enumerate(lanes)}
    for index, lane in enumerate(lanes):
        if index % 4 == 0:
            ax.axhspan(index - .5, index + 3.5, color="#f5f7fa", zorder=0)
    for event in variant["events"]:
        if event["lane"] not in positions:
            continue
        x, dur = event["start"], event["duration"]
        if x + dur < limits[0] or x > limits[1]:
            continue
        color = CATEGORIES[event["category"]][1]
        alpha = .95 if not focus or event["category"] == "kv" else .16
        ax.broken_barh([(x, dur)], (positions[event["lane"]] - .36, .72),
                       facecolors=color, edgecolors="none", alpha=alpha, zorder=3)
    if not focus:
        ax.axhline(31.5, color="#64748b", linewidth=1)
        ax.axvspan(variant["kv_start_us"], variant["kv_end_us"], color=CATEGORIES["kv"][1], alpha=.045)
    tick_step = 8 if compact else 4
    ticks = list(range(0, len(lanes), tick_step))
    ax.set_yticks(ticks, [lanes[i].replace("_", " ") for i in ticks], fontsize=8)
    ax.set_ylim(len(lanes) - .5, -.5)
    ax.set_xlim(*limits)
    ax.grid(axis="x", alpha=.17, zorder=0)
    ax.set_xlabel("相对首个 AICore 任务的时间 / μs")
    ax.tick_params(axis="both", length=0, pad=7)
    for spine in ax.spines.values():
        spine.set_color("#dce1ea")
