"""Archive DSL source-generation and validation evidence for the branch."""
from concurrent.futures import ThreadPoolExecutor
import gzip
import json
import os
from pathlib import Path
import re
import shutil

import zipfile

W = Path(__file__).resolve().parents[2]
S = W / "artifacts/dsv41-semantic-review"
R = W / "pypto-a5-dev/pypto-lib-dsv41-dsl"
D = R / "artifacts/dsv4_1_dsl/2026-11-03"
D.mkdir(parents=True, exist_ok=True)
records = []

def normalized(text):
    text = text.replace(str(W), "${WORKSPACE}")
    text = text.replace("${USER_ROOT}", "${USER_ROOT}")
    text = text.replace("${USER_HOME}", "${USER_HOME}")
    text = re.sub(r"/home/[A-Za-z0-9_.-]+/", "${USER_HOME}/", text)
    text = re.sub(r"/data/disk2/pyptouser/[A-Za-z0-9_.-]+/", "${USER_ROOT}/", text)
    return text

def record(source, target, encoding="copy"):
    records.append(dict(source=normalized(str(source)), path=str(target.relative_to(D)),
                        bytes=target.stat().st_size,
                        encoding=encoding))

def copy(source, target, is_text=True):
    target.parent.mkdir(parents=True, exist_ok=True)
    if is_text:
        target.write_text(normalized(source.read_text()))
    else:
        shutil.copy2(source, target)
    record(source, target, "path-normalized-text" if is_text else "copy")

for f in S.glob("*.log"):
    copy(f, D / "validation" / (f.name + ".txt"))
copy(S / "progress.txt", D / "history/progress.txt")
copy(S / "rsqrt-github-issue-body.txt", D / "issues/rsqrt-3257.txt")

for src_name, dest_name in (("swa-l1-compare", "dsl-vs-cpp"), ("swa-l1-sync-start", "sync-start")):
    src = S / src_name
    dst = D / "swimlanes" / dest_name

            continue

    with zipfile.ZipFile(dst / "swimlanes.zip", "w", compression=zipfile.ZIP_DEFLATED) as z:
        for f in sorted(dst.rglob("*")):
            if f.is_file() and f.suffix != ".zip":
                z.write(f, str(f.relative_to(dst)))

for name, repo in (("cannbotdsl", W / "cannbot-dsl-dev"), ("pypto", W / "issue75-env/pypto-pr")):

# These are frozen model tensors, not a Python installation or a toolchain build.
# Individual gzip members keep every Git object below the hosting size limit.
compress = []
fixture = R / "build_output/_jit_l3_decode_swa_20261103_142957_x9kigqi6"
for f in sorted((fixture / "data").rglob("*.pt")):
    compress.append((f, D / "fixtures/swa-rank1" / f.relative_to(fixture / "data").with_suffix(".pt.gz")))
cpp = W / "pypto-a5-dev/pypto-lib-dsv41-no-dsl/build_output/_jit_l3_decode_swa_20261103_151632_y0edewwk/data/out"
for f in sorted(cpp.glob("*.pt")):
    compress.append((f, D / "fixtures/swa-cpp/out" / (f.name + ".gz")))

    compress.append((f, D / "precision" / (f.name + ".gz")))

compress.append((args_dir / "args.bin", D / "precision/args_dump/args.bin.gz"))

def compress_one(item):
    source, target = item
    target.parent.mkdir(parents=True, exist_ok=True)
    with source.open("rb") as inp, target.open("wb") as raw:
        with gzip.GzipFile(filename="", mode="wb", fileobj=raw, compresslevel=6, mtime=0) as out:
            shutil.copyfileobj(inp, out, length=1024 * 1024)
    if target.stat().st_size >= 95 * 1024 * 1024:
        raise ValueError(f"Oversized compressed member: {target}")
    return dict(source=normalized(str(source)), path=str(target.relative_to(D)),
                raw_bytes=source.stat().st_size,
                bytes=target.stat().st_size, encoding="gzip")
with ThreadPoolExecutor(max_workers=2) as pool:
    records.extend(pool.map(compress_one, compress))
