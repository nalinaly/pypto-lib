"""校验全部 worker 标记、输出保护区及原核逐项一致性。"""

import argparse
import json
from pathlib import Path

import torch
from golden import ratio_allclose, run
from index_route_dsl import ROUTE_CASES, build_route_specs, golden_route, validate_route
from native_route import validate_native_route


work = Path(__file__).resolve().parent
parser = argparse.ArgumentParser()
parser.add_argument("--compile-only", action="store_true")
parser.add_argument("-d", "--device", type=int, default=0)
args = parser.parse_args()
compiled = {} if args.compile_only else {r["label"]: r for r in json.loads((work / "compiled.json").read_text())}
records = []
failures = []


def comparator(label):
    def compare(actual, expected, **kwargs):
        target = work / "device-output" / label
        target.mkdir(parents=True, exist_ok=True)
        torch.save(actual.detach().cpu().clone(), target / "flags.pt")
        return ratio_allclose(atol=0, rtol=0, max_error_ratio=0)(actual, expected, **kwargs)
    return compare


for case in ROUTE_CASES:
    for kind, fn in (("native", validate_native_route), ("dsl", validate_route)):
        label = f"{case}-{kind}"
        print("[候选池判定对照] " + label, flush=True)
        result = run(
            fn=fn,
            specs=build_route_specs(case),
            golden_fn=golden_route,
            config={"backend_type": "a5", "execution_mode": False, "device_id": args.device},
            compare_fn={"flags": comparator(label)},
            compile_only=args.compile_only,
            runtime_dir=None if args.compile_only else compiled[label]["work_dir"],
            save_data=not args.compile_only,
        )
        records.append({"label": label, "passed": bool(result.passed), "error": str(result.error or ""),
                        "work_dir": str(result.work_dir), "compile_only": args.compile_only})
        name = "compiled.json" if args.compile_only else "device-results.json"
        (work / name).write_text(json.dumps(records, ensure_ascii=False, indent=2) + "\n")
        if not result.passed:
            failures.append(label)

if not args.compile_only and not failures:
    comparison = []
    for case in ROUTE_CASES:
        native = torch.load(work / "device-output" / f"{case}-native/flags.pt", weights_only=True)
        dsl = torch.load(work / "device-output" / f"{case}-dsl/flags.pt", weights_only=True)
        different = int((native != dsl).sum())
        comparison.append({"case": case, "不同元素数": different, "完整输出元素数": dsl.numel(),
                           "worker标记": dsl[:32, 0].tolist()})
        if different:
            failures.append(case)
    (work / "native-comparison.json").write_text(json.dumps(comparison, ensure_ascii=False, indent=2) + "\n")
    print("[原核逐项对照] " + json.dumps(comparison, ensure_ascii=False), flush=True)
if failures:
    raise SystemExit("失败用例：" + ", ".join(failures))
