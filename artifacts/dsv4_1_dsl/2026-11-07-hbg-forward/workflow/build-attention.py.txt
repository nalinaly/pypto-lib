"""在 CPU 上装配两种候选池的 HBG 程序。"""
import json
from pathlib import Path
from pypto.ir.distributed_compiled_program import DistributedCompiledProgram
from pypto.runtime.distributed_runner import _assemble_chip_callables
work = Path(__file__).resolve().parent
for label in ('reindex-dense', 'reindex-sparse'):
    rows = json.loads((work / f'unit-{label}-compiled.json').read_text())
    for row in rows:
        assert row['passed'], row
        compiled = DistributedCompiledProgram.from_dir(row['work_dir'])
        _, runtime, _ = _assemble_chip_callables(compiled)
        assert runtime == 'host_build_graph', runtime
    print(label + ' 装配通过', flush=True)
