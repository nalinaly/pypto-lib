"""用固定 EP2 数据和现有整层比较器验证 SWA、C2A，保存实际设备输出。"""

import ast
import json
from pathlib import Path
import sys

import golden
import torch
from golden import ScalarSpec, TensorSpec
import decode_layer


work = Path(__file__).resolve().parent
model = work.parents[1] / "pypto-a5-dev/pypto-lib-dsv41-dsl"
layer = int(sys.argv[sys.argv.index("--layer-id") + 1])
directories = {
    0: "_jit_l3_decode_layer_20261104_135301_3gqotj6h",
    2: "_jit_l3_decode_layer_20261105_160121_u27otki0",
}
fixture = model / "build_output" / directories[layer] / "data"
sys.argv.extend(["--golden-data", str(fixture)])
compile_only = "--compile-only" in sys.argv
original_run = golden.run


def frozen_specs(layer_id=0, seed=0):
    if layer_id != layer or seed != 0:
        raise ValueError("固定数据的层号或种子不匹配")
    specs = []
    for name in decode_layer.l3_decode_layer.param_names:
        path = fixture / "in" / (name + ".pt")
        if not path.is_file():
            path = fixture / "out" / (name + ".pt")
        value = torch.load(path, weights_only=True, mmap=True)
        if value.ndim == 0:
            specs.append(ScalarSpec(name, value.dtype, value.item()))
        else:
            specs.append(TensorSpec(name, list(value.shape), value.dtype))
    return specs


def capture(name, comparator):
    def compare(actual, expected, **kwargs):
        directory = work / "device-output" / f"layer{layer}-ep2"
        directory.mkdir(parents=True, exist_ok=True)
        torch.save(actual.detach().cpu().clone(), directory / f"{name}.pt")
        return comparator(actual, expected, **kwargs)
    return compare


def run(**kwargs):
    kwargs["compare_fn"] = {name: capture(name, compare) for name, compare in kwargs["compare_fn"].items()}
    if not compile_only:
        record = json.loads((work / f"layer{layer}-compile.json").read_text())
        kwargs["runtime_dir"] = record["work_dir"]
    result = original_run(**kwargs)
    record = {"passed": bool(result.passed), "error": str(result.error or ""),
              "work_dir": str(result.work_dir), "compile_only": compile_only,
              "fixture": str(fixture), "layer": layer}
    suffix = "compile" if compile_only else "result"
    (work / f"layer{layer}-{suffix}.json").write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n")
    return result


golden.run = run
namespace = vars(decode_layer).copy()
namespace["build_tensor_specs"] = frozen_specs
source = Path(decode_layer.__file__)
tree = ast.parse(source.read_text())
main = tree.body[-1]
if not isinstance(main, ast.If):
    raise RuntimeError("无法定位整层验证入口")
exec(compile(ast.Module(body=main.body, type_ignores=[]), str(source), "exec"), namespace)
