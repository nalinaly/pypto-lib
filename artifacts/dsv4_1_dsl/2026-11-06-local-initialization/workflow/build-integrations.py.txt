"""在 CPU 编译 MoE 与两条整层路径的 A5 核二进制。"""

import json
from pathlib import Path
from pypto.runtime import device_runner


work = Path(__file__).resolve().parent
results = []
for label in ("moe", "layer0", "layer2"):
    record = json.loads((work / f"{label}-compile.json").read_text())
    if not record["passed"]:
        raise RuntimeError(label + " 前端编译未通过")
    root = Path(record["work_dir"])
    directories = sorted(p.parent for p in (root / "next_levels").rglob("kernel_config.py"))
    for directory in directories:
        print("[集成二进制编译] " + label + " " + str(directory), flush=True)
        device_runner._compile_and_assemble(directory, "a5")
        results.append({"label": label, "work_dir": str(directory), "passed": True})
(work / "integration-binaries.json").write_text(json.dumps(results, ensure_ascii=False, indent=2) + "\n")
print(f"[集成二进制编译] 通过 {len(results)} 组", flush=True)
