"""归档三类本地初始化任务的原核对照、整层验证及设备生成物。"""

import gzip
import json
from pathlib import Path
import re
import shutil
import subprocess


work = Path(__file__).resolve().parent
workspace = work.parents[1]
model = workspace / "pypto-a5-dev/pypto-lib-dsv41-dsl"
archive = model / "artifacts/dsv4_1_dsl/2026-11-06-local-initialization"
inventory = []


def capture(source, target, binary=False):
    source = Path(source)
    destination = archive / target
    destination.parent.mkdir(parents=True, exist_ok=True)
    if binary:
        with source.open("rb") as inp, gzip.open(destination, "wb", compresslevel=6) as out:
            shutil.copyfileobj(inp, out)
    else:
        content = source.read_text().replace(str(workspace), "<workspace>")
        content = content.replace(str(workspace.parent), "<user-home>")
        destination.write_text("\n".join(line.rstrip() for line in content.splitlines()).rstrip() + "\n")
    inventory.append({"文件": str(target), "字节数": destination.stat().st_size})


def generated(label, directory, native=False):
    for name in ("kernel_config.py", "compiled_meta.json", "distributed_meta.json"):
        if (directory / name).is_file():
            capture(directory / name, f"generated/{label}/{name}.txt")
    for source in sorted((directory / "orchestration").glob("*")):
        if source.suffix in (".cpp", ".py"):
            capture(source, f"generated/{label}/orchestration/{source.name}.txt")
    config = directory / "kernel_config.py"
    if config.is_file():
        for name, entry in re.findall(r'"name": "([^"]+)", "source": \'([^\']+)\'', config.read_text()):
            if name in ("zero_bf16_rows", "initialize_routes"):
                for filename in ("entry.cpp", "device_source.hpp", "manifest.json"):
                    capture(Path(entry).parent / filename, f"generated/{label}/foreign/{name}/{filename}.txt")
    if native:
        for source in sorted((directory / "kernels").rglob("*.cpp")):
            capture(source, f"generated/{label}/kernels/{source.name}.txt")


def require_log(label, passes, outputs=None):
    source = Path(f"/tmp/local-init-{label}.log")
    content = source.read_text()
    if len(re.findall(r"^\[RUN\] PASS", content, re.MULTILINE)) != passes:
        raise RuntimeError(label + " 的通过数量不符")
    if outputs is not None and len(re.findall(r"^\[RUN\]   '[^']+' PASS", content, re.MULTILINE)) != outputs:
        raise RuntimeError(label + " 的输出通过数量不符")
    capture(source, f"validation/{label}.log.txt")


records = json.loads((work / "device-results.json").read_text())
comparison = json.loads((work / "native-comparison.json").read_text())

    raise RuntimeError("二十六项核心设备运行必须全部通过")
if len(comparison) != 23 or any(row["不同字节数"] for row in comparison):
    raise RuntimeError("十三组原核对照的二十三个输出必须逐字节一致")
require_log("initial-compile", 13)
require_log("core-compile", 26)
require_log("core-device", 26, 46)
for name in ("binaries", "integration-binaries", "precommit"):
    capture(Path(f"/tmp/local-init-{name}.log"), f"validation/{name}.log.txt")
for name in ("compiled.json", "binary-results.json", "integration-binaries.json", "device-results.json",
             "native-comparison.json", "kernel-inventory.json", "queue-tasks.json", "environment.json"):
    capture(work / name, "validation/" + name)
for name in ("initialization_dsl.py", "hc.py", "gate.py", "decode_layer.py"):
    capture(model / "models/deepseek_v4_1_flash" / name, f"model-snapshot/{name}.txt")
for name in ("hc_baseline.py", "gate_baseline.py", "decode_layer_baseline.py", "native_init.py", "compare-native.py",
             "run-moe.py", "run-layer.py", "run-case.sh", "run-integrations.sh", "run-layers.sh", "build-binaries.py",
             "build-integrations.py", "audit.py", "archive.py"):
    capture(work / name, f"workflow/{name}.txt")
capture(workspace / "artifacts/dsv41-semantic-review/run_case.py", "workflow/run_case.py.txt")
capture(work / "README.txt", "README.txt")
seen = set()
for row in records:
    directory = Path(row["work_dir"])
    if directory in seen:
        continue
    seen.add(directory)
    label = row["label"]
    generated(label, directory, native=label.endswith("-native"))
    for source in sorted((directory / "data").rglob("*.pt")):
        relative = source.relative_to(directory / "data")
        capture(source, f"fixtures/{label}/{relative}.gz", binary=True)

integrations = {}
for label, outputs in (("moe", 3), ("layer0", 16), ("layer2", 16)):
    record = json.loads((work / f"{label}-result.json").read_text())
    if not record["passed"] or record["compile_only"]:
        raise RuntimeError(label + " 集成必须通过")
    require_log(f"{label}-compile", 1)
    require_log(f"{label}-device", 1, outputs)
    for suffix in ("compile", "result"):
        capture(work / f"{label}-{suffix}.json", f"validation/{label}-{suffix}.json")
    root = Path(record["work_dir"])
    generated(label, root)
    for child in sorted((root / "next_levels").glob("*")):
        generated(label + "/" + child.name, child)
    if "fixture" in record:
        fixture = Path(record["fixture"])
        for name in ("layer_id", "attention_num_tokens", "moe_num_tokens"):
            capture(fixture / "in" / f"{name}.pt", f"integration-inputs/{label}/{name}.pt.gz", binary=True)
    integrations[label] = {"状态": "PASS", "输出通过数": outputs}
for source in sorted((work / "device-output").rglob("*.pt")):
    relative = source.relative_to(work / "device-output")
    if relative.parts[0].startswith("layer") and source.stem not in (
        "attention_output", "attention_hidden", "attention_pre_mix", "next_pre_mix", "x_mixed", "x_next"
    ):
        continue
    capture(source, f"device-output/{relative}.gz", binary=True)
