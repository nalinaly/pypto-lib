"""运行既有验证入口，复用编译结果并记录真实比较结果。"""

import argparse
import json
from pathlib import Path
import runpy
import sys

import golden
import golden.runner
import torch


work = Path(__file__).resolve().parent
parser = argparse.ArgumentParser(add_help=False)
parser.add_argument("--label", required=True)
parser.add_argument("--entry", required=True)
args, remaining = parser.parse_known_args()
entry = Path(args.entry).resolve()
compile_only = "--compile-only" in remaining
prefix = work / ("unit-" + args.label)
compiled = [] if compile_only else json.loads(prefix.with_name(prefix.name + "-compiled.json").read_text())
records = []
original_run = golden.run


def capture(index, name, comparator):
    def check(actual, expected, **kwargs):
        target = work / "device-output" / args.label / str(index)
        target.mkdir(parents=True, exist_ok=True)
        torch.save(actual.detach().cpu().clone(), target / f"{name}.pt")
        return comparator(actual, expected, **kwargs)
    return check


def run(*positional, **kwargs):
    index = len(records)
    if not compile_only:
        kwargs["runtime_dir"] = compiled[index]["work_dir"]
    compare = kwargs.get("compare_fn")
    if isinstance(compare, dict):
        kwargs["compare_fn"] = {name: capture(index, name, fn) for name, fn in compare.items()}
    kwargs["save_data"] = True
    result = original_run(*positional, **kwargs)
    records.append({"case": index, "passed": bool(result.passed), "error": str(result.error or ""),
                    "work_dir": str(result.work_dir), "compile_only": compile_only})
    suffix = "compiled" if compile_only else "results"
    prefix.with_name(prefix.name + "-" + suffix + ".json").write_text(
        json.dumps(records, ensure_ascii=False, indent=2) + "\n")
    return result


golden.run = run
golden.runner.run = run
sys.argv = [str(entry), *remaining]
runpy.run_path(str(entry), run_name="__main__")
if not records or not all(row["passed"] for row in records):
    raise SystemExit("入口验证未全部通过：" + args.label)
