"""在 CPU 编译本轮叶子入口的 A5 二进制。"""
import argparse,json
from pathlib import Path
from pypto.runtime import device_runner
work=Path(__file__).resolve().parent
parser=argparse.ArgumentParser()
parser.add_argument('--labels',default='')
parser.add_argument('--cases',default='unit-cases.json')
parser.add_argument('--record',default='unit-binaries.json')
args=parser.parse_args()
selected=set(args.labels.split(',')) if args.labels else None
records=[]
seen=set()
for label,_,_ in json.loads((work/args.cases).read_text()):
    if selected and label not in selected:continue
    cases=json.loads((work/f'unit-{label}-compiled.json').read_text())
    if not all(r['passed'] for r in cases):raise RuntimeError(label+' 前端编译未通过')
    for row in cases:
        root=Path(row['work_dir'])
        directories=([root] if (root/'kernel_config.py').is_file() else [])
        directories+=sorted(p.parent for p in (root/'next_levels').rglob('kernel_config.py'))
        for directory in directories:
            if directory in seen:continue
            seen.add(directory)
            print('[设备二进制] '+label+' '+directory.name,flush=True)
            device_runner._compile_and_assemble(directory,'a5')
            records.append({'label':label,'work_dir':str(directory),'passed':True})
            (work/args.record).write_text(json.dumps(records,ensure_ascii=False,indent=2)+'\n')
print('设备二进制通过 '+str(len(records))+' 组',flush=True)
