"""在 CPU 编译本轮 MoE 的 A5 设备二进制。"""
import json
from pathlib import Path
from pypto.runtime import device_runner
work=Path(__file__).resolve().parent
record=json.loads((work/'moe-compile.json').read_text())
if not record['passed']:
    raise RuntimeError(record['error'])
root=Path(record['work_dir'])/'next_levels/l2_moe'
device_runner._compile_and_assemble(root,'a5')
print('MoE A5 二进制编译通过',flush=True)
