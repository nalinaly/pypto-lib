"""对同一组输入分别运行原 PyPTO 与 DSL C1A 投影，并比较真实设备输出。"""

import argparse
import json
from pathlib import Path

import pypto.language as pl
import torch
from golden import run

from compressor_baseline import c1a_compressor_project as baseline_project
from compressor_dsl import (
    D, HEAD_DIM, build_c1a_projection_specs, golden_c1a_projection,
    projection_comparator, validate_c1a_projection,
)


work = Path(__file__).resolve().parent


@pl.jit
def validate_baseline(
    source: pl.Tensor, weight: pl.Tensor[[D, HEAD_DIM], pl.BF16, pl.NZ],
    output: pl.InOut[pl.Tensor], num_tokens: pl.Scalar[pl.INT32],
):
    rows = pl.tensor.dim(source, 0)
    target = pl.slice(output, [rows, HEAD_DIM], [0, 0])
    ready = pl.system.task_dummy(deps=[])
    baseline_project(source, weight, target, num_tokens, ready)
    return output


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("-d", "--device", type=int, default=0)
    args = parser.parse_args()
    cases = ((17, 15, "layout"), (17, 0, "layout"), (1, 1, "ties"),
             (49, 49, "layout"), (80, 72, "random"), (193, 193, "layout"))
    records = []
    for rows, active, kind in cases:
        record = {"rows": rows, "active": active, "kind": kind, "输出来源": "compare_fn 的真实设备输出"}
        outputs = []
        for label, entry in (("DSL", validate_c1a_projection), ("PyPTO", validate_baseline)):
            captured = {}
            original_compare = projection_comparator(1e-4, 1 / 128) if kind == "random" else projection_comparator(0)

            def capture(actual, expected, **kwargs):
                captured["output"] = actual.detach().cpu().clone()
                return original_compare(actual, expected, **kwargs)

            print(f"[C1A_PROJECTION_COMPARE] {label} {rows=} {active=} {kind=}", flush=True)
            result = run(
                fn=entry,
                specs=build_c1a_projection_specs(rows, active, kind),
                golden_fn=golden_c1a_projection,
                compare_fn={"output": capture},
                config=dict(backend_type="a5", execution_mode=False, device_id=args.device),
                save_data=True,
            )
            if not result.passed:
                raise SystemExit(result.error or 1)
            actual = captured["output"]
            destination = work / "device-output" / f"{rows}-{active}-{kind}" / label
            destination.mkdir(parents=True, exist_ok=True)
            torch.save(actual, destination / "output.pt")
            outputs.append(actual)
            record[label] = str(result.work_dir)
            print(f"[产物] {result.work_dir}", flush=True)
        dsl, native = (value.contiguous().view(torch.uint8) for value in outputs)
        record["不同字节数"] = int((dsl != native).sum())
        record["最大绝对差"] = float((outputs[0].float() - outputs[1].float()).abs().max())
        print(f"[对照统计] {record}", flush=True)
        torch.testing.assert_close(dsl, native, atol=0, rtol=0)
        records.append(record)
        print("[逐字节对照通过]", flush=True)
    (work / "native-comparison.json").write_text(json.dumps(records, ensure_ascii=False, indent=2) + "\n")
