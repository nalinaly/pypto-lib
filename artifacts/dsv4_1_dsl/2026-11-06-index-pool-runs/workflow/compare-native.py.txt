"""校验候选池规划的原核对照及原有请求分组回归。"""

import argparse
import json
from pathlib import Path

import torch
from golden import TensorSpec, ratio_allclose, run
from index_plan_dsl import (
    POOL_PLAN_CASES, build_pool_plan_specs, golden_pool_plan, validate_pool_plan,
    build_plan_specs, golden_plan, validate_plan,
)
from native_plan import validate_native_pool


work = Path(__file__).resolve().parent
parser = argparse.ArgumentParser()
parser.add_argument("--compile-only", action="store_true")
parser.add_argument("-d", "--device", type=int, default=0)
args = parser.parse_args()
compiled = {} if args.compile_only else {r["label"]: r for r in json.loads((work / "compiled.json").read_text())}
records = []
failures = []


def reference_native(tensors):
    """将同一个向量参考计算映射到原核的连续存储布局。"""
    native = tensors["runs"]
    backing = torch.full((5, native.shape[1] + 2), -71, dtype=torch.int32)
    backing[:4, 1:-1] = native
    golden_pool_plan({**tensors, "runs": backing})
    native.copy_(backing[:4, 1:-1])


def comparator(label):
    def compare(actual, expected, **kwargs):
        target = work / "device-output" / label
        target.mkdir(parents=True, exist_ok=True)
        torch.save(actual.detach().cpu().clone(), target / "runs.pt")
        return ratio_allclose(atol=0, rtol=0, max_error_ratio=0)(actual, expected, **kwargs)
    return compare


cases = [(case, kind) for case in POOL_PLAN_CASES for kind in ("native", "dsl")]
cases += [("mixed", "request-0"), ("mixed", "request-1")]
for case, kind in cases:
    label = f"{case}-{kind}"
    if kind.startswith("request"):
        fn, reference = validate_plan, golden_plan
        specs = build_plan_specs(case, int(kind[-1]))
    else:
        fn = validate_native_pool if kind == "native" else validate_pool_plan
        reference = reference_native if kind == "native" else golden_pool_plan
        specs = build_pool_plan_specs(case)
        if kind == "native":
            initial = specs[1].init_value[:4, 1:-1].contiguous()
            specs[1] = TensorSpec("runs", list(initial.shape), torch.int32, init_value=initial)
    print("[候选池规划验证] " + label, flush=True)
    result = run(
        fn=fn,
        specs=specs,
        golden_fn=reference,
        config={"backend_type": "a5", "execution_mode": False, "device_id": args.device},
        compare_fn={"runs": comparator(label)},
        compile_only=args.compile_only,
        runtime_dir=None if args.compile_only else compiled[label]["work_dir"],
        save_data=not args.compile_only,
    )
    records.append({"label": label, "passed": bool(result.passed), "error": str(result.error or ""),
                    "work_dir": str(result.work_dir), "compile_only": args.compile_only})
    name = "compiled.json" if args.compile_only else "device-results.json"
    (work / name).write_text(json.dumps(records, ensure_ascii=False, indent=2) + "\n")
    if not result.passed:
        failures.append(label)

if not args.compile_only and not failures:
    comparison = []
    for case in POOL_PLAN_CASES:
        native = torch.load(work / "device-output" / f"{case}-native/runs.pt", weights_only=True)
        dsl = torch.load(work / "device-output" / f"{case}-dsl/runs.pt", weights_only=True)
        different = int((native != dsl[:4, 1:-1]).sum())
        comparison.append({"case": case, "不同元素数": different, "元数据元素数": native.numel(),
                           "DSL保护区校验": "全部通过零容差 CPU 参考"})
        if different:
            failures.append(case)
    (work / "native-comparison.json").write_text(json.dumps(comparison, ensure_ascii=False, indent=2) + "\n")
    print("[原核逐项对照] " + json.dumps(comparison, ensure_ascii=False), flush=True)
if failures:
    raise SystemExit("失败用例：" + ", ".join(failures))
