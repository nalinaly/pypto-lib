import gzip

inventory = []


def capture(source, target, binary=False):
    source = Path(source)
    destination = archive / target
    destination.parent.mkdir(parents=True, exist_ok=True)
    if binary:
        with source.open("rb") as inp, gzip.open(destination, "wb") as out:
            shutil.copyfileobj(inp, out)
    else:
        text = source.read_text().replace(str(workspace), "<workspace>")
        text = text.replace(str(workspace.parent), "<user-home>")
        destination.write_text("\n".join(line.rstrip() for line in text.splitlines()).rstrip() + "\n")
    inventory.append({"文件": str(target), "字节数": destination.stat().st_size})


def generated(label, directory, fixtures=False):
    for name in ("kernel_config.py", "compiled_meta.json", "distributed_meta.json"):
        if (directory / name).is_file():
            capture(directory / name, f"generated/{label}/{name}.txt")
    for source in sorted((directory / "orchestration").glob("*")):
        if source.suffix in (".cpp", ".py"):
            capture(source, f"generated/{label}/orchestration/{source.name}.txt")
    config = directory / "kernel_config.py"
    if config.is_file():
        for name, source in re.findall(r'"name": "([^"]+)", "source": \'([^\']+)\'', config.read_text()):
            if name == "plan_pool_rows":
                for filename in ("entry.cpp", "device_source.hpp", "manifest.json"):
                    capture(Path(source).parent / filename, f"generated/{label}/foreign/{name}/{filename}.txt")
    if label == "native":
        for source in sorted((directory / "kernels").rglob("*.cpp")):
            capture(source, f"generated/{label}/kernels/{source.name}.txt")
    if fixtures:
        for source in sorted((directory / "data").rglob("*.pt")):
            relative = source.relative_to(directory / "data")
            capture(source, f"fixtures/{label}/{relative}.gz", binary=True)


def require_log(label, passes, outputs=None):
    source = Path(f"/tmp/index-pool-runs-{label}.log")
    content = source.read_text()
    if len(re.findall(r"^\[RUN\] PASS", content, re.MULTILINE)) != passes:
        raise RuntimeError(label + " 的通过数量不符")
    if outputs is not None and len(re.findall(r"^\[RUN\]   '[^']+' PASS", content, re.MULTILINE)) != outputs:
        raise RuntimeError(label + " 的输出通过数量不符")
    capture(source, f"validation/{label}.log.txt")
