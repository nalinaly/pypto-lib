"""运行三条 Attention 入口，保存真实输出并确认 reindex 使用稠密分支。"""

import argparse
import json
from pathlib import Path
import runpy
import sys

import golden
import torch


work = Path(__file__).resolve().parent
workspace = work.parents[1]
model = workspace / "pypto-a5-dev/pypto-lib-dsv41-dsl"
parser = argparse.ArgumentParser()
parser.add_argument("--path", choices=["c1a", "c2a", "reindex"], required=True)
parser.add_argument("--compile-only", action="store_true")
parser.add_argument("-d", "--device", type=int, default=0)
args = parser.parse_args()
entries = {"c1a": "decode_c1a_full.py", "c2a": "decode_c2a_full.py", "reindex": "decode_c1a_reindex.py"}
entry = model / "models/deepseek_v4_1_flash" / entries[args.path]
original_run = golden.run


def capture(name, comparator):
    def compare(actual, expected, **kwargs):
        if name in ("output", "next_pre_mix", "topk_indices", "pool_block_ids", "pool_block_counts"):
            target = work / "device-output" / args.path
            target.mkdir(parents=True, exist_ok=True)
            torch.save(actual.detach().cpu().clone(), target / f"{name}.pt")
        return comparator(actual, expected, **kwargs)
    return compare


def check_dense(directory, cache_pages):
    source = directory / "data/in"
    lengths = torch.load(source / "compressed_lens.pt", weights_only=True)
    counts = torch.load(source / "pool_block_counts.pt", weights_only=True)
    ids = torch.load(source / "pool_block_ids.pt", weights_only=True)
    requests = torch.load(source / "request_ids.pt", weights_only=True)
    table = torch.load(source / "index_block_table.pt", weights_only=True)
    active = int(torch.load(source / "num_tokens.pt", weights_only=True))
    width = ids.shape[1] * 8
    dense = True
    for token in range(active):
        visible = int(lengths[token])
        count = int(counts[token])
        request = int(requests[token])
        dense = dense and count == (visible + 7) // 8 and visible <= min(width, table.shape[1] * 128)
        dense = dense and 0 <= request < table.shape[0]
        dense = dense and torch.equal(ids[token, :count], torch.arange(count, dtype=torch.int32))
        pages = table[request, :(visible + 127) // 128]
        dense = dense and bool(((pages >= 0) & (pages < cache_pages)).all())
    evidence = {"有效Query数": active, "候选池容量": ids.shape[1], "可见长度": lengths[:active].tolist(),
                "候选块数": counts[:active].tolist(), "全部进入稠密分支": dense}
    (work / "reindex-routing.json").write_text(json.dumps(evidence, ensure_ascii=False, indent=2) + "\n")
    if not dense:
        raise RuntimeError("本次 reindex 必须使用含请求规划的稠密分支")


def run(**kwargs):
    kwargs["compare_fn"] = {name: capture(name, compare) for name, compare in kwargs["compare_fn"].items()}
    if not args.compile_only:
        previous = json.loads((work / f"{args.path}-compile.json").read_text())
        kwargs["runtime_dir"] = previous["work_dir"]
    result = original_run(**kwargs)
    if args.path == "reindex" and result.passed and not args.compile_only:
        cache_pages = next(spec.shape[0] for spec in kwargs["specs"] if spec.name == "index_cache")
        check_dense(Path(result.work_dir), cache_pages)
    record = {"passed": bool(result.passed), "error": str(result.error or ""),
              "work_dir": str(result.work_dir), "compile_only": args.compile_only}
    suffix = "compile" if args.compile_only else "result"
    (work / f"{args.path}-{suffix}.json").write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n")
    return result


golden.run = run
sys.argv = [str(entry), "-d", str(args.device)]
if args.path == "reindex":
    sys.argv += ["--start-pos", "4096", "--query-lens", "7"]
if args.compile_only:
    sys.argv += ["--compile-only"]
else:
    sys.argv += ["--save-data"]
runpy.run_path(str(entry), run_name="__main__")
