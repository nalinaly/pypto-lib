"""比较原核与 DSL 的完整 INT32 元数据和所有保护区。"""

import argparse
import json
from pathlib import Path

import torch
from golden import TensorSpec, ratio_allclose, run
from index_plan_dsl import PLAN_CASES, build_plan_specs, golden_plan, validate_plan
from native_plan import validate_native


work = Path(__file__).resolve().parent
parser = argparse.ArgumentParser()
parser.add_argument("--compile-only", action="store_true")
parser.add_argument("-d", "--device", type=int, default=0)
args = parser.parse_args()
compiled = {} if args.compile_only else {row["label"]: row for row in json.loads((work / "compiled.json").read_text())}
records = []
failures = []


def comparator(label):
    def compare(actual, expected, **kwargs):
        target = work / "device-output" / label
        target.mkdir(parents=True, exist_ok=True)
        torch.save(actual.detach().cpu().clone(), target / "runs.pt")
        return ratio_allclose(atol=0, rtol=0, max_error_ratio=0)(actual, expected, **kwargs)
    return compare


def golden_native(tensors):
    """将同一参考计算映射到原核的连续输出布局。"""
    native = tensors["runs"]
    backing = torch.full((5, native.shape[1] + 2), -71, dtype=torch.int32)
    backing[:4, 1:-1] = native
    golden_plan({**tensors, "runs": backing})
    native.copy_(backing[:4, 1:-1])


for case in PLAN_CASES:
    for skip_short in (0, 1):
        for kind, fn in (("native", validate_native), ("dsl", validate_plan)):
            label = f"{case}-{skip_short}-{kind}"
            specs = build_plan_specs(case, skip_short)
            if kind == "native":
                initial = specs[2].init_value[:4, 1:-1].contiguous()
                specs[2] = TensorSpec("runs", list(initial.shape), torch.int32, init_value=initial)
            print(f"[原核对照] {label}", flush=True)
            result = run(
                fn=fn,
                specs=specs,
                golden_fn=golden_native if kind == "native" else golden_plan,
                config={"backend_type": "a5", "execution_mode": False, "device_id": args.device},
                compare_fn={"runs": comparator(label)},
                compile_only=args.compile_only,
                runtime_dir=None if args.compile_only else compiled[label]["work_dir"],
                save_data=not args.compile_only,
            )
            records.append({"label": label, "passed": bool(result.passed), "error": str(result.error or ""),
                            "compile_only": args.compile_only, "work_dir": str(result.work_dir)})
            name = "compiled.json" if args.compile_only else "device-results.json"
            (work / name).write_text(json.dumps(records, ensure_ascii=False, indent=2) + "\n")
            if not result.passed:
                failures.append(label)

if not args.compile_only and not failures:
    comparison = []
    for case in PLAN_CASES:
        for skip_short in (0, 1):
            label = f"{case}-{skip_short}"
            native = torch.load(work / "device-output" / f"{label}-native/runs.pt", weights_only=True)
            dsl = torch.load(work / "device-output" / f"{label}-dsl/runs.pt", weights_only=True)
            logical = dsl[:4, 1:-1]
            different = int((native != logical).sum())
            comparison.append({"case": label, "不同元素数": different, "元数据元素数": native.numel(),
                               "DSL保护区校验": "全部通过零容差 CPU 参考"})
            if different:
                failures.append(label)
    (work / "native-comparison.json").write_text(json.dumps(comparison, ensure_ascii=False, indent=2) + "\n")
    print("[原核逐项对照] " + json.dumps(comparison, ensure_ascii=False), flush=True)
if failures:
    raise SystemExit("失败用例：" + ", ".join(failures))
