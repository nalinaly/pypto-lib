inventory = []


def capture(source, target, binary=False):
    destination = archive / target
    destination.parent.mkdir(parents=True, exist_ok=True)
    if binary:
        with source.open("rb") as inp, gzip.open(destination, "wb") as out:
            shutil.copyfileobj(inp, out)
    else:
        value = source.read_text().replace(str(workspace), "<workspace>")
        value = value.replace(str(workspace.parent), "<user-home>")
        destination.write_text("\n".join(line.rstrip() for line in value.splitlines()).rstrip() + "\n")
    inventory.append({"path": str(destination.relative_to(archive)),
                      "source": str(source.relative_to(workspace)), "bytes": destination.stat().st_size,
                      "encoding": "gzip" if binary else "utf-8"})


def generated(label, directory, fixtures=False):
    for name in ("kernel_config.py", "compiled_meta.json"):
        source = directory / name
        if source.is_file():
            capture(source, f"generated/{label}/{name}.txt")
    for source in sorted((directory / "orchestration").glob("*.cpp")):
        capture(source, f"generated/{label}/orchestration/{source.name}.txt")
    for source in sorted((directory / "kernels").rglob("*.cpp")):
        if "proj_idx_k" in source.name:
            capture(source, f"generated/{label}/kernels/{source.name}.txt")
    config = directory / "kernel_config.py"
    if config.is_file():
        for name, entry in re.findall(r'"name": "([^"]+)", "source": \'([^\']+)\'', config.read_text()):
            if name.startswith(("project_index_key_tile", "project_compressor_tile")):
                for filename in ("entry.cpp", "device_source.hpp", "manifest.json"):
                    capture(Path(entry).parent / filename, f"generated/{label}/foreign/{name}/{filename}.txt")
    if fixtures:
        for source in sorted((directory / "data").rglob("*.pt")):
            capture(source, Path("fixtures") / label / f"{source.relative_to(directory / 'data')}.gz", True)


def require_pass(filename, count):
    text = (work / filename).read_text()
    if len(re.findall(r"^\[RUN\] PASS", text, re.MULTILINE)) != count:
        raise RuntimeError(filename + " 的通过数量不符")
    return text
