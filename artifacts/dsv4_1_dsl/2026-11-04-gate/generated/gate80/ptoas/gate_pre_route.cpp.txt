#include "pto/pto-inst.hpp"
using namespace pto;

enum class PTOAutoSyncTailMode : int {
  kBarrierAll = 0,
  kSetWaitMte3ToSEvent0 = 1,
};

static AICORE inline void ptoas_auto_sync_tail(
    PTOAutoSyncTailMode mode = PTOAutoSyncTailMode::kBarrierAll) {
  switch (mode) {
  case PTOAutoSyncTailMode::kSetWaitMte3ToSEvent0:
    set_flag(PIPE_MTE3, PIPE_S, EVENT_ID0);
    wait_flag(PIPE_MTE3, PIPE_S, EVENT_ID0);
    break;
  case PTOAutoSyncTailMode::kBarrierAll:
  default:
    pipe_barrier(PIPE_ALL);
    break;
  }
}

template <typename Ptr>
static AICORE inline void PTOAS__DCCI_SINGLE_CACHE_LINE(Ptr ptr) {
  dcci((__gm__ void*)ptr, cache_line_t::SINGLE_CACHE_LINE);
}

AICORE void gate_pre_route(__gm__ int32_t* v1, __gm__ float* v2, int64_t v3) {
  using T = float;

  #if defined(__DAV_VEC__)
  set_mask_norm();
  set_vector_mask(-1, -1);
  // pto: %c80_index
  const int64_t v4 = 80;
  // pto: %c6_index
  const int64_t v5 = 6;
  // pto: %c1_index
  const int64_t v6 = 1;
  // pto: %c0_index
  const int64_t v7 = 0;
  // pto: %c0_i32
  const int32_t v8 = 0;
  // pto: %cst_5
  const float v9 = 0.0f;
  for (int64_t v10 = v7; v10 < v4; v10 += v6) {
    // pto: %0
    ;
    if (v3 <= v10) {
      for (int64_t v11 = v7; v11 < v5; v11 += v6) {
        // pto: %flat_offset_mul
        ;
        int64_t v12 = (int64_t) ((uint64_t) v10 * (uint64_t) v5);
        // pto: %flat_offset
        ;
        int64_t v13 = (int64_t) ((uint64_t) v12 + (uint64_t) v11);
        (v1)[v13] = v8;
        (v2)[v13] = v9;
      };
    };
  }
  pipe_barrier(PIPE_ALL);
  dcci((__gm__ void*)0, cache_line_t::ENTIRE_DATA_CACHE);
  dsb((mem_dsb_t)0);
  #endif // __DAV_VEC__

  pipe_barrier(PIPE_ALL);
  dcci((__gm__ void*)0, cache_line_t::ENTIRE_DATA_CACHE);
  dsb((mem_dsb_t)0);
  ptoas_auto_sync_tail(PTOAutoSyncTailMode::kBarrierAll);
  return;
}
