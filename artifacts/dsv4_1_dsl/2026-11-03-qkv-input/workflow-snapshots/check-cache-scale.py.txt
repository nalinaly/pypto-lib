"""CPU check of issue #1666 scale boundaries against an independent reference."""

from pathlib import Path
import sys

import torch

workspace = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(workspace / "pypto-a5-dev/pypto-lib-dsv41-dsl/models/deepseek_v4_1_flash"))
from quant import quantize_mxfp8_cache
from window_kv_dsl import quantize_window_kv

assert quantize_window_kv is quantize_mxfp8_cache
torch.set_num_threads(2)
generator = torch.Generator().manual_seed(1666)
random_rows = torch.randn(2048, 32, generator=generator).bfloat16().float()
thresholds = 448.0 * torch.exp2(torch.tensor([-14.0, -8.0, -1.0, 0.0, 6.0, 100.0]))
boundary = torch.cat((thresholds, torch.nextafter(thresholds, torch.full_like(thresholds, 0)),
                      torch.nextafter(thresholds, torch.full_like(thresholds, float("inf"))),
                      torch.tensor([0.0, 1.0e-4, 1.0])))
boundary_rows = boundary[:, None].expand(-1, 32).clone()
values = torch.cat((random_rows, boundary_rows))
payload, scales = quantize_mxfp8_cache(values)
raw = values.abs().amax(-1).clamp_min(1e-4) * (1.0 / 448.0)
expected_scales = torch.exp2(torch.ceil(torch.log2(raw.double()))).bfloat16()
expected = (values / expected_scales.float()[:, None]).clamp(-448, 448).to(torch.float8_e4m3fn)
torch.testing.assert_close(scales.flatten(), expected_scales, rtol=0, atol=0)
assert torch.equal(payload.view(torch.uint8), expected.view(torch.uint8))
assert scales[-1].item() == 0.00390625
print(f"PASS: {len(values)} groups; payload bytes and BF16 scales exact")
print("PASS: amax=1 -> scale=0.00390625; window DSL golden shares the upstream-aligned quantizer")
print("PASS: zero/clamp cases and adjacent FP32 values around power-of-two thresholds")
